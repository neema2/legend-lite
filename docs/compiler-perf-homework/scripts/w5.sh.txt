CP="$SP/probe-classes:$(cat $SP/cp.txt)"
for r in 1 2 3 4 5; do
  $SP/scripts/run.sh "w5 r$r" $SP/raw/w5.log -- -Xms2g -Xmx2g -Duser.timezone=GMT -Xlog:gc:file=$SP/raw/w5-gc-r$r.log \
    -cp "$CP" com.legend.integration.PerfProbe model $SP/raw/w5-r$r 1 8 $SP/models/h100/model.pure $SP/models/deep.tsv test::RT
done
