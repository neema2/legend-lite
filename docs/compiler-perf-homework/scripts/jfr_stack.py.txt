#!/usr/bin/env python3
"""Throwaway JFR aggregator (docs/compiler-perf-homework, 2026-11-13); a sibling of evidence/compiler/jfr_phases.py.

  jfr_stack.py <jfr tool> <recording.jfr> [--filter REGEX] [--top N]

Differences from jfr_phases.py: a sample is attributed to the INNERMOST legend frame's phase (the first frame from the
top of the stack that names a phase), so a lowering frame called from inside the typer counts as lowering; plan
assembly, PlanJson and the dialect are phases of their own; allocation (jdk.ObjectAllocationSample, by weight) is
attributed the same way. --filter keeps only samples whose stack matches REGEX (e.g. to drop the model build).
Prints markdown: phases (self, by innermost phase) and inclusive; hottest top frames; hottest legend frames
(inclusive); allocation by phase and by top legend frame.
"""
import collections, re, subprocess, sys

PH = [
    ("json", r"com\.legend\.(executionplan\.PlanJson|json\.)"),
    ("render", r"com\.legend\.sql\.dialect\."),
    ("lower", r"com\.legend\.lowering\."),
    ("store-resolve", r"com\.legend\.resolver\."),
    ("inline", r"UserCallInliner|StatementInline"),
    ("type", r"com\.legend\.compiler\.spec\."),
    ("names", r"com\.legend\.compiler\.(NameResolver|BareNames|ResolvedNames|CoreImports)"),
    ("normalize", r"com\.legend\.normalizer\."),
    ("parse", r"com\.legend\.(parser|lexer)\."),
    ("plan", r"com\.legend\.(executionplan|planner|plan)\."),
    ("exec/result", r"com\.legend\.(exec|result|values|test)\."),
    ("model", r"com\.legend\.compiler\.(element\.|KnowledgeLayer|ModelBuilder|ModelIntegrity)|com\.legend\.builtin\."),
]
RX = [(p, re.compile(r)) for p, r in PH]


def stage(frames):
    """The innermost pipeline STAGE on the stack; the model build only when no query stage is on it (a model lookup
    made while lowering is lowering's); helpers (com.legend.model records, com.legend.sql MIR) are their caller's."""
    for x in frames:
        p = phase_of(x)
        if p and p != "model":
            return p
    return "model" if any(phase_of(x) == "model" for x in frames) else "other"


def phase_of(frame):
    for p, rx in RX:
        if rx.search(frame):
            return p
    return None


def events(jfr, f, kind):
    txt = subprocess.run([jfr, "print", "--stack-depth", "96", "--events", kind, f],
                         capture_output=True, text=True).stdout
    for ev in txt.split(kind + " {")[1:]:
        frames = [f"{c}.{m}" for c, m in re.findall(r"^\s+([a-zA-Z_$][\w.$]*)\.([\w$<>]+)\(", ev, re.M)]
        w = re.search(r"weight = ([\d.]+) ([kMG]?B)", ev)
        weight = 1
        if w:
            weight = float(w.group(1)) * {"B": 1, "kB": 1e3, "MB": 1e6, "GB": 1e9}[w.group(2)]
        yield frames, weight


def main():
    jfr, f = sys.argv[1], sys.argv[2]
    flt = None
    top_n = 30
    if "--filter" in sys.argv:
        flt = re.compile(sys.argv[sys.argv.index("--filter") + 1])
    if "--top" in sys.argv:
        top_n = int(sys.argv[sys.argv.index("--top") + 1])
    selfp = collections.Counter(); inclp = collections.Counter(); top = collections.Counter()
    incl = collections.Counter(); n = 0
    for frames, _ in events(jfr, f, "jdk.ExecutionSample"):
        if not frames:
            continue
        stack = "\n".join(frames)
        if flt and not flt.search(stack):
            continue
        n += 1
        top[frames[0]] += 1
        ph = stage(frames)
        selfp[ph] += 1
        for p in {phase_of(x) for x in frames} - {None}:
            inclp[p] += 1
        for x in dict.fromkeys(x for x in frames if x.startswith("com.legend")):
            incl[x] += 1
    print(f"execution samples: {n}\n\n| phase | innermost | share | inclusive | share |\n|---|---|---|---|---|")
    for p, c in selfp.most_common():
        print(f"| {p} | {c} | {100 * c / n:.1f}% | {inclp[p]} | {100 * inclp[p] / n:.1f}% |")
    print("\n| hottest frame at the top of the stack | samples | share |\n|---|---|---|")
    for x, c in top.most_common(15):
        print(f"| `{x}` | {c} | {100 * c / n:.1f}% |")
    print("\n| hottest legend frame, inclusive | samples | share |\n|---|---|---|")
    for x, c in incl.most_common(top_n):
        print(f"| `{x}` | {c} | {100 * c / n:.1f}% |")
    ap = collections.Counter(); af = collections.Counter(); tot = 0
    for frames, w in events(jfr, f, "jdk.ObjectAllocationSample"):
        stack = "\n".join(frames)
        if not frames or (flt and not flt.search(stack)):
            continue
        tot += w
        ap[stage(frames)] += w
        lf = next((x for x in frames if x.startswith("com.legend")), "(none)")
        af[lf] += w
    if tot:
        print(f"\nallocation (sampled weight): {tot / 1e9:.2f} GB\n\n| phase | share |\n|---|---|")
        for p, c in ap.most_common():
            print(f"| {p} | {100 * c / tot:.1f}% |")
        print("\n| allocating legend frame (innermost) | share |\n|---|---|")
        for x, c in af.most_common(20):
            print(f"| `{x}` | {100 * c / tot:.1f}% |")


main()
