"""Read a Bazel test.log written with LEGEND_LITE_DUMP_SQL=1; print each SENT statement once (unique, sorted), with
run-specific text normalized: the execution id, sandbox and temp paths, UUIDs. Output: gzip of NUL-separated stmts."""
import gzip, re, sys
log, out = sys.argv[1], sys.argv[2]
UUID = re.compile(r'[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}')
PATHS = [re.compile(r'/root/\.cache/bazel/[^\s\'"]*'), re.compile(r'/tmp/[^\s\'"]*'), re.compile(r'junit-?\d+')]
HEAD = re.compile(r'^\[sql:[a-z_]+(?: execution=[0-9a-f-]+)?\] ')
stmts, cur = set(), None
def flush():
    global cur
    if cur is not None:
        s = '\n'.join(cur).rstrip()
        s = UUID.sub('<UUID>', s)
        for p in PATHS:
            s = p.sub('<PATH>', s)
        stmts.add(s)
    cur = None
import io
with (io.TextIOWrapper(sys.stdin.buffer, encoding='utf-8', errors='replace') if log == '-' else open(log, encoding='utf-8', errors='replace')) as f:
    for line in f:
        line = line.rstrip('\n')
        m = HEAD.match(line)
        if m:
            flush()
            cur = [m.group(0).split(' execution=')[0].rstrip('] ') + '] ' + line[m.end():]]
        elif cur is not None:
            if line.startswith('[') or line.startswith("'[") or line.startswith('JUnit') or line.startswith('╔') or line.startswith('==='):
                flush()
            else:
                cur.append(line)
flush()
with gzip.open(out, 'wt', encoding='utf-8') as g:
    g.write('\0'.join(sorted(stmts)))
print(len(stmts))
