# THE WALL MEASUREMENT (2026-11-06, the user: "this IS the invariant we need to ship"): report-only marks on every
# boundary, applied to the working tree for one measurement run and NOT committed as is.
#   [wall-touch]  a database touch (open, check, send, load, metadata, transaction) outside a plan's run
#   [run-wall]    a planning act (type, inline, resolve, lower, render) while a plan runs (the existing report mode)
#   [wall-nested] a run started inside a run
# Enabled by -Dlegend.wall.measure=true; every mark only reports, so the patched tree behaves as the head (the strict
# guards keep refusing): one run both verifies the head and measures the wall.
import re

root = './'   # run from the repository's root


def edit(path, pairs):
    p = root + path
    s = open(p).read()
    for old, new in pairs:
        assert s.count(old) == 1, (path, old[:80])
        s = s.replace(old, new)
    open(p, 'w').write(s)


def guard_after_signature(path, signature_prefixes, line):
    """Insert `line` as the first statement of each method whose signature line starts with a prefix."""
    p = root + path
    src = open(p).read().split('\n')
    out = []
    pending = 0
    hits = 0
    for l in src:
        out.append(l)
        stripped = l.strip()
        if pending == 0 and any(stripped.startswith(sp) for sp in signature_prefixes):
            pending = 1
        if pending == 1 and l.rstrip().endswith('{'):
            indent = re.match(r'^(\s*)', l).group(1) + '    '
            out.append(indent + line)
            pending = 0
            hits += 1
    open(p, 'w').write('\n'.join(out))
    return hits

    /** The nearest callers outside {@code skipPrefix} and base, for the measurement's report. */
    private static String outside(String skipPrefix, int depth) {
        StringBuilder out = new StringBuilder();
        int named = 0;
        for (StackTraceElement f : new Throwable().getStackTrace()) {
            String c = f.getClassName();
            if (c.startsWith(skipPrefix) || c.startsWith("com.legend.base.") || c.startsWith("java.")
                    || c.startsWith("jdk.") || c.startsWith("sun.")) {
                continue;
            }
            if (named++ == depth) {
                break;
            }
            out.append(out.length() == 0 ? "" : " <- ").append(c.substring(c.lastIndexOf('.') + 1)).append('.')
                    .append(f.getMethodName()).append(':').append(f.getLineNumber());
        }
        return out.toString();
    }
