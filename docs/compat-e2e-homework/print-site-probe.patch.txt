+  *  statement prints a value as text. OUTPUT: a root projection IS the print of one value (a Java presenter could

+ final class PrintSiteProbe {
+     private static final ThreadLocal<Boolean> IN = ThreadLocal.withInitial(() -> false);
+     private static final String DIR = System.getenv("TEST_UNDECLARED_OUTPUTS_DIR") != null
+             ? System.getenv("TEST_UNDECLARED_OUTPUTS_DIR") : System.getProperty("legend.diag.printSites");

+     private PrintSiteProbe() {
+     }

+     static boolean on() {
+         return System.getProperty("legend.diag.printSites") != null && !IN.get();
+     }

+     static boolean isPrint(SqlExpr e) {
+         if (e instanceof SqlExpr.Cast c) {
+             return c.target() == SqlType.Scalar.VARCHAR;
+         }
+         if (e instanceof SqlExpr.Call c) {
+             return switch (c.fn().name()) {
+                 case "FLOAT_PRINT", "FLOAT_CANON", "BOOL_TO_TEXT", "DECIMAL_TO_TEXT", "STRFTIME" -> true;
+                 default -> false;
+             };
+         }
+         return false;
+     }

+     static final class Found {
+         final List<SqlExpr> output = new ArrayList<>();
+         final Map<String, int[]> fns = new java.util.TreeMap<>();
+         int nested;
+         int inQuery;
+         final Map<String, Integer> ctx = new java.util.TreeMap<>();
+ 
+         void count(SqlExpr e, int bucket) {
+             String k = e instanceof SqlExpr.Call c ? c.fn().name() : "CAST_VARCHAR";
+             fns.computeIfAbsent(k, x -> new int[3])[bucket]++;
+         }
+     }

+     static void query(SqlQuery q, boolean root, Found f) {
+         if (q instanceof SqlSelect s) {
+             for (SqlSelect.Projection p : s.projections()) {
+                 if (isPrint(p.expr())) {
+                     if (root) {
+                         f.output.add(p.expr());
+                         f.count(p.expr(), 0);
+                     } else {
+                         f.nested++;
+                         f.count(p.expr(), 1);
+                     }
+                     walk(childrenOf(p.expr()), f);
+                 } else {
+                     walk(p.expr(), f);
+                 }
+             }
+             walk(s.from(), f);
+             PARENTS.push("WHERE");
+             walk(s.where(), f);
+             PARENTS.pop();
+             PARENTS.push("GROUPBY");
+             walk(s.groupBy(), f);
+             PARENTS.pop();
+             PARENTS.push("HAVING");
+             walk(s.having(), f);
+             walk(s.qualify(), f);
+             PARENTS.pop();
+             PARENTS.push("ORDERBY");
+             walk(s.orderBy(), f);
+             PARENTS.pop();
+             return;
+         }
+         if (q instanceof com.legend.sql.SqlUnion u) {
+             for (Object o : components(u)) {
+                 if (o instanceof SqlQuery b) {
+                     query(b, root, f);
+                 } else if (o instanceof List<?> l) {
+                     for (Object x : l) {
+                         if (x instanceof SqlQuery b) {
+                             query(b, root, f);
+                         } else {
+                             walk(x, f);
+                         }
+                     }
+                 } else {
+                     walk(o, f);
+                 }
+             }
+             return;
+         }
+         if (q instanceof com.legend.sql.SqlWith w) {
+             for (Object o : components(w)) {
+                 if (o instanceof SqlQuery b) {
+                     query(b, root, f);
+                 } else {
+                     walk(o, f);
+                 }
+             }
+             return;
+         }
+         walk(components(q), f);
+     }

+     private static List<@com.legend.base.Nullable Object> childrenOf(SqlExpr e) {
+         return components(e);
+     }

+     private static List<@com.legend.base.Nullable Object> components(Object o) {
+         List<@com.legend.base.Nullable Object> out = new ArrayList<>();
+         if (o instanceof Record r) {
+             for (var c : r.getClass().getRecordComponents()) {
+                 try {
+                     out.add(c.getAccessor().invoke(r));
+                 } catch (ReflectiveOperationException ex) {
+                     throw new IllegalStateException(ex);
+                 }
+             }
+         }
+         return out;
+     }

+     private static final java.util.ArrayDeque<String> PARENTS = new java.util.ArrayDeque<>();

+     private static String label(Object o) {

+             return c.fn().name();

+         return o.getClass().getSimpleName();

+     static void walk(@com.legend.base.Nullable Object o, Found f) {
+         if (o == null || o instanceof String || o instanceof Number || o instanceof Boolean || o instanceof Enum<?>) {
+             return;
+         }
+         if (o instanceof SqlExpr.RaiseText) {
+             return;
+         }
+         if (o instanceof SqlQuery q) {
+             query(q, false, f);
+             return;
+         }
+         if (o instanceof SqlExpr e && isPrint(e)) {
+             f.inQuery++;
+             f.count(e, 2);
+             String operand = e instanceof SqlExpr.Cast cc ? label(cc.value()) : "";
+             f.ctx.merge(label(e) + "<-" + (PARENTS.isEmpty() ? "?" : PARENTS.peek()) + "|" + operand, 1, Integer::sum);
+             walk(components(e), f);
+             return;
+         }
+         if (o instanceof Iterable<?> it) {
+             for (Object x : it) {
+                 walk(x, f);
+             }
+             return;
+         }
+         if (o instanceof java.util.Optional<?> opt) {
+             opt.ifPresent(x -> walk(x, f));
+             return;
+         }
+         if (o instanceof Map<?, ?> m) {
+             m.values().forEach(x -> walk(x, f));
+             return;
+         }
+         if (o instanceof Record) {
+             PARENTS.push(label(o));
+             try {
+                 walk(components(o), f);
+             } finally {
+                 PARENTS.pop();
+             }
+         }
+     }

+     static void record(AnsiSqlRenderer dialect, SqlQuery q, String sql) {
+         Found f = new Found();
+         try {
+             query(q, true, f);
+         } catch (RuntimeException ex) {
+             line(dialect, sql, "WALKFAIL\t" + ex);
+             return;
+         }
+         int without = sql.length();
+         String fail = "";
+         if (!f.output.isEmpty()) {
+             IdentityHashMap<SqlExpr, Boolean> leaves = new IdentityHashMap<>();
+             f.output.forEach(e -> leaves.put(e, true));
+             SqlQuery stripped = strip(q, leaves);
+             IN.set(true);
+             try {
+                 without = dialect.render(stripped).length();
+             } catch (RuntimeException ex) {
+                 fail = "STRIPFAIL";
+             } finally {
+                 IN.set(false);
+             }
+         }
+         StringBuilder fns = new StringBuilder();
+         f.fns.forEach((k, v) -> fns.append(k).append('=').append(v[0]).append('/').append(v[1]).append('/')
+                 .append(v[2]).append(';'));
+         fns.append('\t');
+         f.ctx.forEach((k, v) -> fns.append(k).append('=').append(v).append(';'));
+         line(dialect, sql, f.output.size() + "\t" + f.nested + "\t" + f.inQuery + "\t" + sql.length() + "\t"
+                 + without + "\t" + fail + "\t" + fns);
+     }

+     private static SqlQuery strip(SqlQuery q, IdentityHashMap<SqlExpr, Boolean> leaves) {
+         if (q instanceof SqlSelect s) {
+             List<SqlSelect.Projection> ps = new ArrayList<>();
+             for (SqlSelect.Projection p : s.projections()) {
+                 SqlExpr e = p.expr();
+                 if (leaves.containsKey(e)) {
+                     e = e instanceof SqlExpr.Cast c ? c.value() : ((SqlExpr.Call) e).args().get(0);
+                 }
+                 ps.add(e == p.expr() ? p : new SqlSelect.Projection(e, p.alias(), p.out()));
+             }
+             return new SqlSelect(ps, s.distinct(), s.from(), s.where(), s.groupBy(), s.having(), s.qualify(),
+                     s.orderBy(), s.limit(), s.offset(), s.outputs());
+         }
+         if (q instanceof com.legend.sql.SqlUnion) {
+             return new com.legend.sql.SqlRewriter() {
+                 @Override
+                 protected SqlQuery select(SqlSelect s) {
+                     return strip(s, leaves);
+                 }
+             }.rewrite(q);
+         }
+         return q;
+     }

+             java.nio.file.Files.writeString(p, d.getClass().getName() + "\t" + Integer.toHexString(sql.hashCode())
+                     + "\t" + rest + "\n", java.nio.file.StandardOpenOption.CREATE,
+                     java.nio.file.StandardOpenOption.APPEND);
+         } catch (java.io.IOException ex) {
+             throw new java.io.UncheckedIOException(ex);
+         }
+     }
