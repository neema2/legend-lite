QUERY_KW = {'SELECT', 'WITH', 'VALUES', 'TABLE', 'SHOW', 'DESCRIBE'}
EFFECT_NATIVES = ['executeInDb', 'dropAndCreateTableInDb', 'dropAndCreateSchemaInDb', 'loadCsvToDbTable', 'createTempTable', 'dropTempTable']

ASSERT_RE = re.compile(r'^assert\w*$')
HEAD_RE = re.compile(r'^\s*((?:[A-Za-z_]\w*::)*)([A-Za-z_]\w*)\s*\(')
LET_RE = re.compile(r'^let\s+([A-Za-z_]\w*)\s*=\s*(.*)$', re.S)
CALL_RE = re.compile(r'(?<![\w$:])((?:[A-Za-z_]\w*::)*)([A-Za-z_]\w*)\s*\(')
ARROW_CALL_RE = re.compile(r'->\s*((?:[A-Za-z_]\w*::)*)([A-Za-z_]\w*)\s*\(')

def first_arg(s, i):
    """s[i] == '(' ; returns the first argument's text."""
    e = match_balanced(s, i, '(', ')')
    return split_top(s[i+1:e-1], ',')[0].strip(), s[i+1:e-1], e

def literal_single_query(arg):
    a = arg.strip()
    if not (a.startswith("'") and a.endswith("'") and a.count("'") - a.count("\\'") == 2):
        return None  # not one literal
    sql = a[1:-1]
    stmts = [x for x in re.split(r';', sql) if x.strip()]
    if len(stmts) != 1:
        return False
    s = stmts[0].strip()
    s = re.sub(r'^(--[^\n]*\n|\s)+', '', s)
    m = re.match(r'[A-Za-z]+', s)
    return bool(m) and m.group(0).upper() in QUERY_KW

def executeInDb_calls(text):
    """Every executeInDb( / executeInDbToTDS( call in text: (pos, name, first-arg, kind)"""
    out = []
    for m in re.finditer(r'(?<![\w$])(executeInDb(?:ToTDS)?)\s*\(', text):
        name = m.group(1)
        i = m.end() - 1
        arg, args, end = first_arg(text, i)
        q = literal_single_query(arg)
        kind = 'grid' if q else ('effect:non-literal-sql' if q is None else 'effect:non-query-literal')
        out.append({'pos': m.start(), 'name': name, 'arg': arg, 'kind': kind, 'end': end})
    return out

def head(s):
    m = HEAD_RE.match(s)
    return (m.group(1), m.group(2)) if m else (None, None)

def classify(stmts, helpers_effect, caller=None):
    """Per statement: kind + flags; the driver's precedence (effect, let, assert, value)."""
    res = []
    n = len(stmts)
    for i, s in enumerate(stmts):
        last = i == n - 1
        calls = executeInDb_calls(s)
        direct_effect = any(c['kind'] != 'grid' for c in calls) or any(re.search(r'(?<![\w$])' + e + r'\s*\(', s) for e in EFFECT_NATIVES[1:])
        prints = bool(re.search(r'(?<![\w$>])(println|print)\s*\(', s))
        helper_effect = None
        if callable(helpers_effect):
            for pm in CALL_RE.finditer(s):
                r = helpers_effect(pm.group(2), caller)
                if r: helper_effect = r; break
        flags = set()
        if PIVOT in s:
            tail = s[s.rfind(PIVOT):]
            flags.add('pivot-uncast' if CAST not in tail else 'pivot-cast')
        if ALL_RE.search(s): flags.add('all()')
        if re.search(r'(?<![\w$])getAll\s*\(', s): flags.add('getAll(')
        if any(re.search(r'(?<![\w$])' + t + r'\s*\(', s) for t in TDG_NATIVES): flags.add('tdg')
        if re.search(r'->\s*stage\s*\(', s): flags.add('stage')
        kind = None; sub = None; letname = None; rhs = None
        lm = LET_RE.match(s)
        if direct_effect:
            kind = 'effect'; sub = 'direct'
        elif helper_effect:
            kind = 'effect'; sub = 'via-helper'; flags.add('via:' + '>'.join(x.split('::')[-1] for x in helper_effect))
        elif prints and not lm:
            kind = 'effect'; sub = 'print'
        elif lm and not last:
            letname, rhs = lm.group(1), lm.group(2).strip()
            if re.fullmatch(r'\$[A-Za-z_]\w*', rhs):
                kind = 'let:alias'
            elif 'stage' in flags:
                kind = 'let:staged'
            else:
                pk, hn = head(rhs)
                if hn in FRAME_NATIVES:
                    kind = 'let:frame'
                elif hn in ('executeInDb', 'executeInDbToTDS') and calls and calls[0]['kind'] == 'grid' and calls[0]['pos'] <= len(s) - len(rhs) + 2:
                    kind = 'let:raw-grid'
                elif hn is not None and hn.startswith('execute'):
                    kind = 'let:value'; sub = 'helper-execute*:' + hn
                else:
                    kind = 'let:value'
        elif lm and last:
            kind = 'value'; sub = 'trailing-let'
        else:
            pk, hn = head(s)
            if hn and ASSERT_RE.match(hn):
                kind = 'assert'; sub = hn
            elif any(ASSERT_RE.match(c.group(2)) for c in CALL_RE.finditer(s)) or any(ASSERT_RE.match(c.group(2)) for c in ARROW_CALL_RE.finditer(s)):
                kind = 'assert'; sub = 'nested:' + (hn or s[:12])
            else:
                kind = 'value'
        res.append({'i': i, 'last': last, 'kind': kind, 'sub': sub, 'flags': sorted(flags), 'let': letname, 'text': s})
    return res

def platform_owned(fqn):
    return fqn in PLATFORM or fqn.startswith('meta::pure::functions::')
SUBSUMED = {'meta::relational::extension::relationalExtensions', 'meta::relational::functions::sqlQueryToString::createDbConfig'}
def helper_index(funcs):
    """A resolver: reaches(simple name, caller) -> (bool, path). Definitions keyed by simple name; a call is resolved to the
    definitions in the caller's own package or its file's imports (else any definition of that name: a flagged fallback).
    Platform-owned twins (the catalog's natives, meta::pure::functions::*) and the two Subsumed FQNs are excluded."""
    by = collections.defaultdict(list)
    for f in funcs:
        if platform_owned(f['name']) or f['name'] in SUBSUMED:
            continue
        by[f['name'].split('::')[-1]].append(f)
    memo = {}
    def candidates(name, caller):
        defs = by.get(name, [])
        scoped = [d for d in defs if d['pkg'] == caller['pkg'] or d['pkg'] in caller['imports']]
        return scoped if scoped else defs
    def direct(body):
        calls = executeInDb_calls(body)
        return any(c['kind'] != 'grid' for c in calls) or any(re.search(r'(?<![\w$])' + e + r'\s*\(', body) for e in EFFECT_NATIVES[1:])
    def reaches_def(d, stack):
        key = d['name'] + '@' + str(d['line'])
        if key in memo: return memo[key]
        if key in stack: return None
        stack.add(key)
        r = None
        if direct(d['body']): r = [d['name']]
        else:
            for m in CALL_RE.finditer(d['body']):
                nm = m.group(2)
                if nm == d['name'].split('::')[-1]: continue
                for c in candidates(nm, d):
                    sub = reaches_def(c, stack)
                    if sub: r = [d['name']] + sub; break
                if r: break
        stack.discard(key); memo[key] = r; return r
    def reaches(name, caller):
        for c in candidates(name, caller):
            r = reaches_def(c, set())
            if r: return r
        return None
    reaches.by = by
    return reaches

def frame_reads_before_end(cl):
    out = []
    n = len(cl)
    for st in cl:
        if st['kind'] == 'let:frame':
            v = '$' + st['let']
            readers = [t['i'] for t in cl if t['i'] > st['i'] and not t['last'] and re.search(re.escape(v) + r'(?![\w])', t['text'])]
            if readers: out.append((st['let'], st['i'] + 1, [r + 1 for r in readers]))
    return out

def census(label, bodies, helpers_effect):
    kinds_b = collections.Counter(); kinds_s = collections.Counter(); asserts = collections.Counter()
    flags_b = collections.Counter(); flags_s = collections.Counter(); subs = collections.Counter()
    h6 = []; h1 = []; framereads = []; stmts_total = 0; lets_helper = collections.Counter()
    uncast_any = []; pivot_bodies = []
    percls = {}
    for b in bodies:
        stmts = split_statements(b['body'])
        cl = classify(stmts, helpers_effect, b)
        percls[b['name']] = cl
        stmts_total += len(cl)
        seen = set(); fseen = set()
        for st in cl:
            kinds_s[st['kind']] += 1; seen.add(st['kind'])
            if st['kind'] == 'assert': asserts[st['sub'] if not st['sub'].startswith('nested') else st['sub']] += 1
            if st['sub'] and st['sub'].startswith('helper-execute'): lets_helper[st['sub']] += 1
            if st['kind'] == 'effect': subs['effect:' + st['sub']] += 1
            for f in st['flags']:
                if f.startswith('via:'): continue
                flags_s[f] += 1; fseen.add(f)
            if 'pivot-uncast' in st['flags'] and not st['last']:
                h6.append((b['name'], st['i'] + 1, len(cl)))
            if 'pivot-uncast' in st['flags']:
                tail = st['text'][st['text'].rfind(PIVOT):]
                if re.search(r'(->\s*)?(toString|toCSV|toJSON)\s*\(', tail):
                    h1.append((b['name'], st['i'] + 1, len(cl)))
        for k in seen: kinds_b[k] += 1
        for f in fseen: flags_b[f] += 1
        fr = frame_reads_before_end(cl)
        if fr: framereads.append((b['name'], fr))
        if any('pivot-uncast' in st['flags'] or 'pivot-cast' in st['flags'] for st in cl): pivot_bodies.append(b['name'])
    return {'label': label, 'bodies': len(bodies), 'statements': stmts_total, 'kinds_by_bodies': kinds_b, 'kinds_by_statements': kinds_s,
            'asserts': asserts, 'flags_by_bodies': flags_b, 'flags_by_statements': flags_s, 'effect_subs': subs, 'lets_helper_execute': lets_helper,
            'h6': h6, 'h1': h1, 'framereads': framereads, 'pivot_bodies': pivot_bodies, 'percls': percls}
