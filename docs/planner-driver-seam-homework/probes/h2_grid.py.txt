def chain_after(text, pos):
    """navigation text after pos up to a depth-0 ',' ')' ';' or end."""
    depth = 0; i = pos; n = len(text)
    while i < n:
        c = text[i]
        if c == "'":
            i += 1
            while i < n and text[i] != "'":
                if text[i] == '\\': i += 1
                i += 1
        elif c in '([{': depth += 1
        elif c in ')]}':
            if depth == 0: break
            depth -= 1
        elif c in ',;' and depth == 0: break
        i += 1
    return text[pos:i]

def norm(s):
    s = re.sub(r"'(?:[^'\\]|\\.)*'", "'S'", s)
    s = re.sub(r'\$[A-Za-z_]\w*', '$V', s)
    s = re.sub(r'\b\d+\b', 'N', s)
    return re.sub(r'\s+', '', s)

def classes_of(uses, stmt_ctx):
    cls = set(); shapes = []
    for u in uses:
        nu = norm(u)
        shapes.append(nu)
        if re.search(r'\.rows(->at\(N\))?\.(value|get\w*)\(|\.rows->map\(V\|\$V\.(value|get\w*)\(|\.rows(->at\(N\))?\.values->at\(|\.rows->map\(V\|\$V\.values->at\(|->at\(N\)\.values->at\(', nu.replace('(V|', '(V|')):
            cls.add('a')
        if '.columnNames' in nu: cls.add('b')
        if re.search(r'\.values(?!->at\()', nu) or re.search(r'->map\(V\|\$V\.values', nu): cls.add('c')
        if re.search(r'\.rows->size\(|->size\(|->isEmpty\(|->isNotEmpty\(', nu): cls.add('d')
    for u, ctx in zip(uses, stmt_ctx):
        nu = norm(u); nctx = norm(ctx)
        if re.search(r'assert(Size|Empty|NotEmpty)\(' , nctx) and re.search(r'^(\.rows)?$', nu): cls.add('d')
        elif re.search(r'^(\.rows)?$', nu) and re.search(r'(print|println)\(', nctx): cls.add('e:print')
        elif re.search(r'^(\.rows)?$', nu) and re.search(r'[A-Za-z_]\w*\(\s*[^;]*\$V', nctx): cls.add('e:helper-arg')
        elif re.search(r'^->(cast|resultSetToTDS)\(', nu) or nu.startswith('.connection') or nu.startswith('->'): cls.add('g:' + nu[:40])
    return cls, shapes

def run(label, files):
    sites = []
    for fpath in files:
        for f in functions(fpath):
            body = f['body']; stmts = split_statements(body)
            # statement spans in body
            spans = []; cursor = 0
            for s in stmts:
                k = body.find(s, cursor); spans.append((k, k + len(s))); cursor = k + len(s)
            for c in executeInDb_calls(body):
                si = next((i for i, (a, b) in enumerate(spans) if a <= c['pos'] < b), None)
                stmt = stmts[si] if si is not None else ''
                lm = LET_RE.match(stmt)
                letv = None; uses = []; ctx = []
                direct_chain = chain_after(body, c['end'])
                if lm:
                    rhs = lm.group(2).strip()
                    if re.match(r'(?:[A-Za-z_]\w*::)*' + c['name'] + r'\s*\(', rhs):
                        letv = lm.group(1)
                        if direct_chain.strip(): uses.append(direct_chain); ctx.append(stmt)
                        for j in range(si + 1, len(stmts)):
                            for m in re.finditer(r'\$' + re.escape(letv) + r'(?![\w])', stmts[j]):
                                uses.append(chain_after(stmts[j], m.end())); ctx.append(stmts[j])
                if letv is None:
                    if direct_chain.strip(): uses.append(direct_chain); ctx.append(stmt)
                cls, shapes = classes_of(uses, ctx)
                statement_level = not lm and si is not None and stmt.lstrip().startswith(c['name']) and not direct_chain.strip()
                nested_in_let_rhs = bool(lm) and letv is None
                if not uses and not cls:
                    if statement_level: cls.add('g:result-discarded(statement-level call)')
                    elif nested_in_let_rhs or (not lm and not statement_level): cls.add('g:argument-of-another-call,no-navigation')
                    else: cls.add('g:let-bound,never-read')
                line = pathlib.Path(fpath).read_text(encoding='utf-8', errors='replace').count('\n', 0, 0) + f['bodyline'] + body.count('\n', 0, c['pos'])
                sites.append({'input': label, 'file': fpath.split('/src/main/resources/')[1], 'line': line, 'fn': f['name'], 'role': role(f),
                              'native': c['name'], 'typer': c['kind'], 'let': letv, 'classes': sorted(cls), 'shapes': shapes, 'stmt_last': si == len(stmts) - 1 if si is not None else None})
    return sites
