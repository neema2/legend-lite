// B3's second measurement (2026-11-11, the B3 worker): DuckDB 1.5.6 JDBC -- (1) every S3 secret field and the scope
// bound as parameters (KEY_ID, SECRET, SESSION_TOKEN, REGION, ENDPOINT, URL_STYLE, USE_SSL as a boolean or its text,
// SCOPE); (2) the server's fence as statements: allowed_directories, enable_external_access=false, lock_configuration=true,
// their order, their reach (database-wide) and whether the folders bind. Probe values only, no real key.
// Run: javac -cp duckdb_jdbc.jar FenceProbe.java Fence2.java && java -cp .:duckdb_jdbc.jar FenceProbe (then Fence2).
// Also measured the same day, in the repository's own tests: DuckDB-WASM 1.33.1-dev57 makes the same secret from bound
// values (//engine-client:node_duckdb_test); its listing, like JDBC's, redacts SECRET and SESSION_TOKEN and shows KEY_ID.
// An S3 secret needs httpfs: offline, JDBC's autoload fails ("Extension Autoloading Error ... httpfs").

import java.sql.*;
import java.nio.file.*;
public class FenceProbe {
  static void step(Connection c, String sql, Object... params) {
    try (PreparedStatement p = c.prepareStatement(sql)) {
      for (int i = 0; i < params.length; i++) p.setObject(i + 1, params[i]);
      boolean rs = p.execute();
      String out = "";
      if (rs) { try (ResultSet r = p.getResultSet()) { StringBuilder sb = new StringBuilder(); int n = r.getMetaData().getColumnCount(); while (r.next()) { sb.append("["); for (int i=1;i<=n;i++) sb.append(r.getString(i)).append(i<n?", ":""); sb.append("]"); } out = sb.toString(); } }
      System.out.println("OK   " + sql + " " + java.util.Arrays.toString(params) + " " + out);
    } catch (SQLException e) {
      System.out.println("FAIL " + sql + " " + java.util.Arrays.toString(params) + " -> " + e.getMessage().split("\n")[0]);
    }
  }
  public static void main(String[] a) throws Exception {
    Path fence = Files.createTempDirectory("fence");
    Path outside = Files.createTempDirectory("outside");
    Files.writeString(fence.resolve("in.csv"), "a,b\n1,2\n");
    Files.writeString(outside.resolve("out.csv"), "a,b\n3,4\n");
    try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
      System.out.println("== secrets, every field bound");
      step(c, "CREATE OR REPLACE TEMPORARY SECRET \"s1\" (TYPE S3, KEY_ID ?, SECRET ?, SESSION_TOKEN ?, REGION ?, ENDPOINT ?, URL_STYLE ?, USE_SSL ?, SCOPE ?)",
          "SENTINEL_KEY", "SENTINEL_SECRET", "SENTINEL_TOKEN", "eu-west-1", "localhost:9000", "path", false, "s3://bucket/");
      step(c, "SELECT name, type, persistent, scope FROM duckdb_secrets()");
      step(c, "SELECT secret_string LIKE '%SENTINEL_SECRET%' FROM duckdb_secrets()");
      step(c, "CREATE OR REPLACE TEMPORARY SECRET \"s1\" (TYPE S3, USE_SSL ?, SCOPE ?)", "true", "s3://bucket/");
      step(c, "CREATE OR REPLACE TEMPORARY SECRET \"s1\" (TYPE S3, USE_SSL ?, SCOPE ?)", true, "s3://bucket/");
      step(c, "DROP TEMPORARY SECRET IF EXISTS \"s1\"");
      step(c, "DROP TEMPORARY SECRET IF EXISTS \"s1\"");
      System.out.println("== the fence, as statements");
      step(c, "SET TimeZone = 'UTC'");
      step(c, "SET allowed_directories = ?", (Object) new String[]{fence.toString()});
      step(c, "SELECT current_setting('allowed_directories')");
      step(c, "SET allowed_directories = ['" + fence + "']");
      step(c, "SELECT current_setting('allowed_directories')");
      step(c, "SET enable_external_access = false");
      step(c, "SET lock_configuration = true");
      step(c, "SELECT count(*) FROM read_csv(?)", fence.resolve("in.csv").toString());
      step(c, "SELECT count(*) FROM read_csv(?)", outside.resolve("out.csv").toString());
      step(c, "SET TimeZone = 'UTC'");
      step(c, "SET allowed_directories = ['/']");
      step(c, "SET enable_external_access = true");
      step(c, "ATTACH ':memory:' AS other");
      step(c, "INSTALL httpfs");
    }
    System.out.println("== a second connection to one database");
    try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
      step(c, "SET allowed_directories = ['" + fence + "']");
      step(c, "SET enable_external_access = false");
      step(c, "SET lock_configuration = true");
      try (Connection d = ((org.duckdb.DuckDBConnection) c).duplicate()) {
        step(d, "SELECT count(*) FROM read_csv(?)", outside.resolve("out.csv").toString());
        step(d, "SELECT count(*) FROM read_csv(?)", fence.resolve("in.csv").toString());
        step(d, "SET TimeZone = 'UTC'");
        step(d, "SET allowed_directories = ['" + fence + "']");
        step(d, "SET enable_external_access = false");
      }
    }
    System.out.println("== order: external access off first, then the folders");
    try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
      step(c, "SET enable_external_access = false");
      step(c, "SET allowed_directories = ['" + fence + "']");
      step(c, "SELECT count(*) FROM read_csv(?)", fence.resolve("in.csv").toString());
    }
  }
}

import java.sql.*;
import java.nio.file.*;
public class Fence2 {
  public static void main(String[] a) throws Exception {
    Path fence = Files.createTempDirectory("fence");
    Files.writeString(fence.resolve("in.csv"), "a,b\n1,2\n");
    String[] sqls = {"SET allowed_directories = ?", "SET GLOBAL allowed_directories = ?", "SET allowed_directories = [?]", "SET allowed_directories = ?::VARCHAR[]", "SELECT set_config('allowed_directories', ?)", "CALL set_config('allowed_directories', ?)", "SET VARIABLE fence = ?", "SET allowed_directories = getvariable('fence')", "SET allowed_directories = (SELECT getvariable('fence'))"};
    for (String sql : sqls) {
      for (int kind = 0; kind < 3; kind++) {
        try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
          if (sql.contains("getvariable")) { try (PreparedStatement p = c.prepareStatement("SET VARIABLE fence = ?")) { p.setObject(1, kind == 0 ? fence.toString() : c.createArrayOf("VARCHAR", new Object[]{fence.toString()})); p.execute(); } catch (SQLException e) { System.out.println("  var: " + e.getMessage().split("\n")[0]); } }
          try (PreparedStatement p = c.prepareStatement(sql)) {
            if (sql.contains("?")) { if (kind == 0) p.setString(1, fence.toString()); else if (kind == 1) p.setObject(1, c.createArrayOf("VARCHAR", new Object[]{fence.toString()})); else p.setString(1, "['" + fence + "']"); }
            p.execute();
            try (ResultSet r = c.createStatement().executeQuery("SELECT current_setting('allowed_directories')")) { r.next(); System.out.println("OK   " + sql + " kind" + kind + " -> " + r.getString(1)); }
          } catch (SQLException e) { System.out.println("FAIL " + sql + " kind" + kind + " -> " + e.getMessage().split("\n")[0]); }
        }
        if (!sql.contains("?")) break;
      }
    }
  }
}
