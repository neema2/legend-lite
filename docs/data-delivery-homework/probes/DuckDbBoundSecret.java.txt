// B3's first measurement (2026-11-11): whether DuckDB's CREATE TEMPORARY SECRET takes its credential's values as bound
// parameters, so exec binds them from its secret store and no statement text, and no plan, holds a value.
// Run: javac -cp duckdb_jdbc.jar SecretProbe.java && java -cp .:duckdb_jdbc.jar SecretProbe (probe values only, no real key).
import java.sql.*;

public class SecretProbe {
    public static void main(String[] a) throws Exception {
        try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
            try (Statement s = c.createStatement(); ResultSet r = s.executeQuery("SELECT version()")) { r.next(); System.out.println("duckdb " + r.getString(1)); }
            String[] forms = {
                "CREATE TEMPORARY SECRET probe_s (TYPE S3, KEY_ID ?, SECRET ?, REGION 'us-east-1')",
                "CREATE OR REPLACE TEMPORARY SECRET probe_s (TYPE S3, KEY_ID ?, SECRET ?, SCOPE ?)",
            };
            for (String f : forms) {
                try (PreparedStatement ps = c.prepareStatement(f)) {
                    int n = ps.getParameterMetaData().getParameterCount();
                    for (int i = 1; i <= n; i++) ps.setString(i, i == 3 ? "s3://probe-bucket" : "probe-value-" + i);
                    ps.execute();
                    System.out.println("BOUND ok (" + n + " params): " + f);
                } catch (SQLException e) {
                    System.out.println("BOUND refused: " + f + " :: " + e.getMessage().split("\n")[0]);
                }
            }
            try (Statement s = c.createStatement(); ResultSet r = s.executeQuery("SELECT name, type, persistent, storage, scope FROM duckdb_secrets()")) {
                while (r.next()) System.out.println("secret: " + r.getString(1) + " " + r.getString(2) + " persistent=" + r.getString(3) + " storage=" + r.getString(4) + " scope=" + r.getString(5));
            } catch (SQLException e) { System.out.println("duckdb_secrets: " + e.getMessage().split("\n")[0]); }
            try (Statement s = c.createStatement()) {
                s.execute("CREATE TEMPORARY SECRET lit_s (TYPE S3, KEY_ID 'k', SECRET 'v')");
                System.out.println("literal ok");
                s.execute("DROP TEMPORARY SECRET lit_s");
                System.out.println("drop temporary ok");
            } catch (SQLException e) { System.out.println("literal: " + e.getMessage().split("\n")[0]); }
        }
    }
}
