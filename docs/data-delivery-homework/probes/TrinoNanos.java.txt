        return "(CAST(date_diff('second', TIMESTAMP '1970-01-01 00:00:00', date_trunc('second', CAST(" + t

            + " AS TIMESTAMP(9)) AS VARCHAR), 21, 9) AS DECIMAL(38, 0)))";
    }
    static String fromNanosDec(String x) {

    }

import java.sql.*;
public class TrinoNs3 {
    static String micros(String t) { return "(date_diff('second', TIMESTAMP '1970-01-01 00:00:00', date_trunc('second', CAST(" + t + " AS TIMESTAMP(9)))) * 1000000 + CAST(substr(CAST(CAST(" + t + " AS TIMESTAMP(9)) AS VARCHAR), -9, 6) AS BIGINT))"; }
    static String microsOld(String t) { return "(date_diff('second', TIMESTAMP '1970-01-01 00:00:00', date_trunc('second', CAST(" + t + " AS TIMESTAMP(6)))) * 1000000 + CAST(substr(CAST(CAST(" + t + " AS TIMESTAMP(6)) AS VARCHAR), 21, 6) AS BIGINT))"; }
    static String nanos(String t) { return "(CAST(date_diff('second', TIMESTAMP '1970-01-01 00:00:00', date_trunc('second', CAST(" + t + " AS TIMESTAMP(9)))) AS DECIMAL(38, 0)) * 1000000000 + CAST(substr(CAST(CAST(" + t + " AS TIMESTAMP(9)) AS VARCHAR), -9) AS DECIMAL(38, 0)))"; }
    static String fromNanos(String x) { return "CAST(at_timezone(from_unixtime_nanos(" + x + "), 'UTC') AS TIMESTAMP(9))"; }
    public static void main(String[] a) throws Exception {
        try (Connection c = DriverManager.getConnection("jdbc:trino://127.0.0.1:18089/memory/default?user=legend&timezone=UTC");
             Statement s = c.createStatement()) {
            s.execute("DROP TABLE IF EXISTS t6");
            s.execute("CREATE TABLE t6 (ts TIMESTAMP(6))");
            s.execute("INSERT INTO t6 VALUES (TIMESTAMP '1969-12-31 23:59:59.500000'), (TIMESTAMP '2024-01-02 03:04:05.123456')");
            String[][] q = {
                {"micros new vs old (col)", "SELECT array_agg(" + micros("ts") + " - " + microsOld("ts") + ") FROM t6"},
                {"micros ns lit", "SELECT " + micros("TIMESTAMP '2024-01-02 03:04:05.123456789'")},
                {"add 3us col", "SELECT array_agg(CAST(" + fromNanos(nanos("ts") + " + CAST(3 AS DECIMAL(38, 0)) * 1000") + " AS VARCHAR)) FROM t6"},
                {"add -2us ns lit", "SELECT CAST(" + fromNanos(nanos("TIMESTAMP '2024-01-02 03:04:05.000000001'") + " + CAST(-2 AS DECIMAL(38, 0)) * 1000") + " AS VARCHAR)"},
                {"strftime micro of ns", "SELECT substr(CAST(CAST(TIMESTAMP '2024-01-02 03:04:05.123456789' AS TIMESTAMP(9)) AS VARCHAR), -9, 6)"},
                {"strftime nano of ns", "SELECT substr(CAST(CAST(TIMESTAMP '2024-01-02 03:04:05.123456789' AS TIMESTAMP(9)) AS VARCHAR), -9)"},
                {"strftime micro of col", "SELECT array_agg(substr(CAST(CAST(ts AS TIMESTAMP(9)) AS VARCHAR), -9, 6)) FROM t6"},
                {"tz of col (9)", "SELECT array_agg(CAST(CAST(at_timezone(with_timezone(ts, 'UTC'), 'Asia/Kolkata') AS TIMESTAMP(9)) AS VARCHAR)) FROM t6"},
                {"compare 9 to 6 col", "SELECT count(*) FROM t6 WHERE ts < TIMESTAMP '2024-01-02 03:04:05.123456001'"},
            };
            for (String[] x : q) {
                try (ResultSet r = s.executeQuery(x[1])) {
                    r.next();
                    System.out.println(x[0] + " :: " + r.getString(1));
                } catch (SQLException e) {
                    System.out.println(x[0] + " :: ERROR " + e.getMessage().split("\n")[0]);
                }
            }
        }
    }
}
