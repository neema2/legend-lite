// session at the B checkpoint's lane round). Usage: java -cp <the BigQuery JDBC driver>:. BigQueryDataTableProbe '<jdbc url>' [user] [password]
// What it asks, in order, each answer printed (a refusal printed, never hidden):
//   1. a USER's permanent table of the session table's name, holding the user's row;
//   2. the session table made as lite would make it (below), two records filled -- a quote and a line break in their
//      JSON text -- and read back ordered by their place;
//   3. the session table dropped as lite would drop it, then the user's table read: its row must be there;
//   4. the drop run again where the session holds no table of the name: it must raise nothing and drop nothing of
//      the user's; and a second connection must not see the session's table while it lives.
// The SQL is the candidate spelling, written from the database's documentation, NOT measured: the probe's output
// decides SqlDialect.dataTable for BigQuery, or its refusal by name (a register row).
// a BigQuery TEMP table exists only inside a script or a session (the connection must be opened with createSession=true): the probe shows whether the driver's statements share one; the user's table needs a dataset (legend_lite_probe) the lane's project holds.
import java.sql.*;

public class BigQueryDataTableProbe {
    static final String USER_TABLE = "CREATE TABLE legend_lite_probe.legend_data_probe (data STRING)";
    static final String[] MAKE = new String[] {"CREATE TEMP TABLE legend_data_probe (data JSON, u_frame_ord__ INT64)"};
    static final String[] INSERT = new String[] {"INSERT INTO legend_data_probe VALUES (JSON '{\"t\":\"it\\\\'s\"}', 0), (JSON '{\"t\":\"a\\\\\\\\nb\"}', 1)"};
    static final String READ = "SELECT TO_JSON_STRING(data), u_frame_ord__, JSON_VALUE(data, '$.t') FROM legend_data_probe ORDER BY u_frame_ord__";
    static final String DROP = "DROP TABLE IF EXISTS legend_data_probe";

    public static void main(String[] a) throws Exception {
        try (Connection c = a.length > 1 ? DriverManager.getConnection(a[0], a[1], a.length > 2 ? a[2] : null)
                : DriverManager.getConnection(a[0]); Statement s = c.createStatement()) {
            step("1 the user's table", s, USER_TABLE);
            step("1 the user's row", s, USER_TABLE.contains("legend_lite_probe.")
                    ? "INSERT INTO legend_lite_probe.legend_data_probe VALUES ('the user''s row')"
                    : "INSERT INTO legend_data_probe VALUES ('the user''s row')");
            for (String m : MAKE) step("2 make", s, m);
            for (String i : INSERT) step("2 fill", s, i);
            query("2 read", s, READ);
            try (Connection other = a.length > 1 ? DriverManager.getConnection(a[0], a[1], a.length > 2 ? a[2] : null)
                    : DriverManager.getConnection(a[0]); Statement o = other.createStatement()) {
                query("4 another connection reads it (must be refused)", o, READ);
            }
            step("3 drop", s, DROP);
            query("3 the user's table", s, USER_TABLE.contains("legend_lite_probe.")
                    ? "SELECT data FROM legend_lite_probe.legend_data_probe" : "SELECT data FROM legend_data_probe");
            step("4 drop again (the session holds none)", s, DROP);
            query("4 the user's table again", s, USER_TABLE.contains("legend_lite_probe.")
                    ? "SELECT data FROM legend_lite_probe.legend_data_probe" : "SELECT data FROM legend_data_probe");
        }
    }

    static void step(String what, Statement s, String sql) {
        try {
            s.execute(sql);
            System.out.println(what + " :: ok :: " + sql);
        } catch (SQLException e) {
            System.out.println(what + " :: REFUSED " + e.getMessage().split("\n")[0] + " :: " + sql);
        }
    }

    static void query(String what, Statement s, String sql) {
        try (ResultSet r = s.executeQuery(sql)) {
            StringBuilder rows = new StringBuilder();
            int n = r.getMetaData().getColumnCount();
            while (r.next()) {
                rows.append('[');
                for (int i = 1; i <= n; i++) {
                    rows.append(i > 1 ? " | " : "").append(r.getString(i));
                }
                rows.append(']');
            }
            System.out.println(what + " :: " + rows);
        } catch (SQLException e) {
            System.out.println(what + " :: REFUSED " + e.getMessage().split("\n")[0]);
        }
    }
}
