import java.sql.*;
public class GridCol {
    public static void main(String[] a) throws Exception {
        try (Connection c = DriverManager.getConnection(a[0]); Statement s = c.createStatement()) {
            s.executeUpdate("CREATE TABLE g (s VARCHAR(32))"); s.executeUpdate("INSERT INTO g VALUES ('Credit'), ('FX')"); StringBuilder out = new StringBuilder();
            for (String str : new String[]{"Credit", "FX"})
                for (int st = -8; st <= 3; st++) {
                    for (int len = -1; len <= 10; len++) {
                        String v;
                        try (ResultSet r = s.executeQuery("SELECT SUBSTRING(s, " + st + ", " + len + ") FROM g WHERE s = '" + str + "'")) { r.next(); v = r.getString(1); }
                        catch (SQLException e) { v = "ERR"; }
                        out.append(str).append(' ').append(st).append(' ').append(len).append(' ').append(v).append('\n');
                    }
                    String v;
                    try (ResultSet r = s.executeQuery("SELECT SUBSTRING(s, " + st + ") FROM g WHERE s = '" + str + "'")) { r.next(); v = r.getString(1); }
                    catch (SQLException e) { v = "ERR"; }
                    out.append(str).append(' ').append(st).append(" - ").append(v).append('\n');
                }
            System.out.print(out);
        }
    }
}
