"""H11: two corpus host passes compared (base, spike), one lane: rosters, the verdict ledger, the FAIL lines, and
the spike's [H11] measurement lines. Usage: compare.py <base dir> <spike dir> <lane>"""
import sys, collections, re
base, spike, lane = sys.argv[1], sys.argv[2], sys.argv[3]
def lines(d, f):
    try: return open(f"{d}/{f}").read().splitlines()
    except FileNotFoundError: return None
print(f"== lane {lane}")
for f in ["fail-roster", "skipped-roster", "unordered-register", "engine-order-register"]:
    a, b = lines(base, f"{lane}-{f}.txt"), lines(spike, f"{lane}-{f}.txt")
    if a is None or b is None: print(f"  {f}: missing"); continue
    sa, sb = set(a), set(b)
    print(f"  {f}: base {len(a)} spike {len(b)} {'IDENTICAL' if a == b else 'DIFFERS'}")
    for x in sorted(sa - sb): print(f"    - {x}")
    for x in sorted(sb - sa): print(f"    + {x}")
a, b = lines(base, "judge-host.tsv"), lines(spike, "judge-host.tsv")
print(f"  judge-host.tsv: base {len(a)} rows spike {len(b)} rows {'BYTE-IDENTICAL' if a == b else 'DIFFERS'}")
if a != b:
    ka = {tuple(l.split('\t')[:2]): l for l in a}; kb = {tuple(l.split('\t')[:2]): l for l in b}
    for k in sorted(set(ka) | set(kb)):
        if ka.get(k) != kb.get(k): print(f"    {k[0]} #{k[1]}: {ka.get(k, '(none)').split(chr(9))[-1]} -> {kb.get(k, '(none)').split(chr(9))[-1]}")
def fails(d):
    m = {}
    for l in lines(d, "host.log"):
        mm = re.match(r"\[corpus2\] FAIL (\S+) :: (.*)", l)
        if mm: m[mm.group(1)] = mm.group(2)
    return m
fa, fb = fails(base), fails(spike)
changed = [t for t in sorted(set(fa) | set(fb)) if fa.get(t) != fb.get(t)]
print(f"  FAIL lines: base {len(fa)} spike {len(fb)}; tests whose failure moved or whose words changed: {len(changed)}")
for t in changed:
    print(f"    {t}\n      base:  {(fa.get(t) or 'PASS')[:300]}\n      spike: {(fb.get(t) or 'PASS')[:300]}")
for l in lines(spike, "host.log"):
    if re.match(r"\[corpus2\] (pass=|.*pass=)", l): print("  spike", l.strip()[:200])
for l in lines(base, "host.log"):
    if re.match(r"\[corpus2\] (pass=|.*pass=)", l): print("  base ", l.strip()[:200])
# the spike's measurement lines
h = [l for l in lines(spike, "host.log") if l.startswith("[H11")]
kinds = collections.Counter()
for l in h:
    p = l.split()
    if p[0] == "[H11]":
        if p[1] in ("HELD", "DIFFERS"):
            forms = next((x for x in p if x.startswith("forms=")), "")
            how = next((x for x in p if x.startswith("how=")), "")
            main = next((x for x in p if x.startswith("main=")), "")
            f = forms.replace("forms=", "")
            f = re.sub(r":[^|]*", ":<sep>", f)
            kinds[(p[1], how, f, main)] += 1
        else:
            kinds[(p[1], p[-1] if p[1] in ("REACHED", "BUILDER") else "")] += 1
    else:
        kinds[(p[0], p[1] if len(p) > 1 else "")] += 1
print("  [H11] lines:")
for k, v in sorted(kinds.items(), key=lambda x: (-x[1], x[0])): print(f"    {v:5d} {' '.join(k)}")
