public class Sub {
    public static void main(String[] a) {
        String s = "abcdef";
        int[][] cases = {{-1, -2}, {7, -2}, {0, 9}, {4, 2}, {-1, 3}};
        System.out.println("# " + System.getProperty("java.version"));
        for (int[] c : cases) {
            try { System.out.println((c[1] == -2 ? "substring(" + c[0] + ")" : "substring(" + c[0] + ", " + c[1] + ")") + "\t" + (c[1] == -2 ? s.substring(c[0]) : s.substring(c[0], c[1]))); }
            catch (RuntimeException e) { System.out.println((c[1] == -2 ? "substring(" + c[0] + ")" : "substring(" + c[0] + ", " + c[1] + ")") + "\t" + e.getClass().getName() + ": " + e.getMessage()); }
        }
    }
}
