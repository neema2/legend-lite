+        private List<String> h13Rows(List<List<Object>> rows) {
+            List<String> out = new ArrayList<>();
+            for (List<Object> r : rows) {
+                StringBuilder b = new StringBuilder();
+                for (Object c : r) {
+                    Object v = c;
+                    if (v instanceof Number num && !(v instanceof Integer || v instanceof Long)) {
+                        v = new java.math.BigDecimal(num.toString()).stripTrailingZeros().toPlainString();
+                    } else if (v instanceof Integer || v instanceof Long) {
+                        v = v.toString();
+                    }
+                    b.append(v).append('|');
+                }
+                out.add(b.toString());
+            }
+            java.util.Collections.sort(out);
+            return out;
+        }
