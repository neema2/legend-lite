import java.sql.*;
/** H8 (j.4), census B2, 2026-11-19: H2's own answers to the four H2 PCT error cases' unchecked SQL, on the H2 a JVM's
 *  class path carries (lite's 2.5.252 or legend-engine's 2.1.214), in legend-engine's MODE=LEGACY. */
public class H8j_H2Plain {
    public static void main(String[] a) throws Exception {
        Class.forName("org.h2.Driver");
        String[][] q = {
            {"5->rem(0)", "SELECT MOD(5, 0)"},
            {"sqrt(-1.0)", "SELECT SQRT(-1.0)"},
            {"1->bitShiftLeft(63)", "SELECT BITSHIFTLEFT(CAST(1 AS BIGINT), 63)"},
            {"1->bitShiftRight(63)", "SELECT BITSHIFTRIGHT(CAST(1 AS BIGINT), 63)"},
            {"1->bitShiftLeft(63) (LSHIFT)", "SELECT LSHIFT(CAST(1 AS BIGINT), 63)"},
            {"1->bitShiftRight(63) (RSHIFT)", "SELECT RSHIFT(CAST(1 AS BIGINT), 63)"},
        };
        try (Connection c = DriverManager.getConnection("jdbc:h2:mem:q;MODE=LEGACY;DEFAULT_NULL_ORDERING=HIGH", "sa", "")) {
            System.out.println("# H2 " + c.getMetaData().getDatabaseProductVersion());
            for (String[] x : q) {
                String out;
                try (Statement s = c.createStatement(); ResultSet r = s.executeQuery(x[1])) {
                    r.next();
                    out = "VALUE " + r.getString(1);
                } catch (SQLException e) {
                    out = "ERROR " + e.getMessage().split("\n")[0];
                }
                System.out.println(x[0] + "\t" + x[1] + "\t" + out);
            }
        }
    }
}
