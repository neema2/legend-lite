"""H8 (k), census B2, 2026-11-19: legend-engine's EXPECTED FAILURES in the relational corpus, run on lite.

The corpus honours legend-engine's own exclusion marks (PureTests: ToFix, ExcludeAlloy), so these 148 tests never run
on main. The probe (H8k_probe.patch.txt) runs them and nothing else. This script joins:
  - the tests and their marks, scanned as MinimalCorpusTest.pinCensus scans them (comments stripped);
  - each lane's outcome from the probe's pass: its fail roster, its skipped roster, and its "[corpus2] FAIL" line;
  - the test body's asserts (the TEST'S OWN EXPECTED VALUES, written by legend-engine's authors), from the judge ledger.
usage: python3 -I H8k_classify.py <engine root> <probe bazel-bin/spec> <out tsv>
"""
import pathlib, re, sys, collections
root = pathlib.Path(sys.argv[1]) / ("legend-engine-xts-relationalStore/legend-engine-xt-relationalStore-generation/"
    "legend-engine-xt-relationalStore-pure/legend-engine-xt-relationalStore-core-pure/src/main/resources/core_relational/relational")
spec = pathlib.Path(sys.argv[2])
block = re.compile(r"<<([^>]*)>>\s*(?:\{[^}]*\}\s*)?([\w:]+)\s*\(")
tests = {}
for f in sorted(root.rglob("*.pure")):
    src = re.sub(r"(?s)/\*.*?\*/", "", f.read_text(errors="replace"))
    src = re.sub(r"//.*", "", src)
    for m in block.finditer(src):
        st = m.group(1)
        if re.search(r"\btest\.Test\b", st) and re.search(r"\b(ToFix|ExcludeAlloy)\b", st):
            marks = "+".join(x for x in ("ToFix", "ExcludeAlloy", "AlloyOnly") if re.search(r"\b" + x + r"\b", st))
            body_start = m.end()
            nxt = block.search(src, body_start)
            body = src[body_start: nxt.start() if nxt else len(src)]
            trivial = re.match(r"\)\s*:\s*Boolean\[1\]\s*\{\s*true\s*;?\s*\}", body) is not None
            tests[m.group(2)] = (marks, str(f.relative_to(root)), trivial)
print(len(tests), "tests marked ToFix or ExcludeAlloy", file=sys.stderr)
lanes = {}
for lane in ("h2", "duckdb"):
    d = spec / ("judge_host_" + lane)
    fail = set(l.split()[0] for l in (d / (lane + "-fail-roster.txt")).read_text().splitlines() if l.strip())
    skip = set(l.split()[0] for l in (d / (lane + "-skipped-roster.txt")).read_text().splitlines() if l.strip())
    why = {}
    for l in (d / "host.log").read_text(errors="replace").splitlines():
        m = re.match(r"\[corpus2\] FAIL (\S+) :: (.*)", l)
        if m:
            why[m.group(1)] = m.group(2)
    asserts = collections.defaultdict(list)
    for l in (d / "judge-host.tsv").read_text().splitlines():
        p = l.split("\t")
        if len(p) >= 4:
            asserts[p[0]].append(p[2].split("::")[-1] + ":" + p[3])
    lanes[lane] = (fail, skip, why, asserts)

def cls(t, lane):
    fail, skip, why, asserts = lanes[lane]
    if t in skip:
        return "skipped", ""
    if t not in fail:
        return "pass", ",".join(sorted(set(asserts.get(t, []))))
    w = why.get(t, "")
    # a SQL-text assert whose text failed and whose rows appeal declined: legend-engine's text decided, not an answer
    if "oracle declined" in w or "(sql-text, DB2" in w or "(sql-text, Composite" in w or "rows underivable" in w:
        return "fail:text", w
    # an answer compared with the test's expected value (or the golden's rows) and differing
    if w.startswith("AssertFailed") or "ROW verdict" in w or "diverged" in w:
        return "fail:answer", w
    return "fail:refused", w

rows = []
for t in sorted(tests):
    marks, f, trivial = tests[t]
    r = [t, marks, "trivial" if trivial else ""]
    for lane in ("h2", "duckdb"):
        c, w = cls(t, lane)
        r += [c, w.replace("\t", " ")[:300]]
    r.append(f)
    rows.append(r)
pathlib.Path(sys.argv[3]).write_text("test\tmarks\ttrivial\th2\th2_detail\tduckdb\tduckdb_detail\tfile\n"
                                     + "\n".join("\t".join(r) for r in rows) + "\n")
c = collections.Counter((r[1], r[2], r[3], r[5]) for r in rows)
for k, v in sorted(c.items(), key=lambda x: -x[1]):
    print(v, k)
