import java.sql.*;
/** H8 (j), census B2, 2026-11-19: DuckDB 1.5.6's own answer to pure/v1's raise.divideByZero (year / 0) unchecked. */
public class H8j_DuckDbDivide {
    public static void main(String[] a) throws Exception {
        Class.forName("org.duckdb.DuckDBDriver");
        try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
            for (String q : new String[] {"SELECT 2023 / 0", "SELECT CAST(2023 AS DOUBLE) / CAST(0 AS DOUBLE)", "SELECT sqrt(-2.0)"}) {
                try (Statement s = c.createStatement(); ResultSet r = s.executeQuery(q)) {
                    r.next();
                    System.out.println(q + "\tVALUE " + r.getString(1));
                } catch (SQLException e) {
                    System.out.println(q + "\tERROR " + e.getMessage().split("\n")[0]);
                }
            }
        }
    }
}
