import java.sql.*;
/** H8 (j), census B2, 2026-11-19: DuckDB's own answers to the six PCT error cases' unchecked SQL (what lite's plain
 *  set and legend-engine both leave to the database), so the ordinary-function rewrite can be judged against them. */
public class H8j_DuckDbPlain {
    public static void main(String[] a) throws Exception {
        String[][] q = {
            {"slice([1,2,3,4,5], 3, 2)", "SELECT list_slice([1,2,3,4,5], 4, 2)"},
            {"rem(5, 0)", "SELECT 5 % 0"},
            {"range(0, 5, 0)", "SELECT range(0, 5, 0)"},
            {"sqrt(-1.0)", "SELECT sqrt(-1.0)"},
            {"1->bitShiftRight(63)", "SELECT 1::BIGINT >> 63"},
            {"1->bitShiftLeft(63)", "SELECT 1::BIGINT << 63"},
        };
        Class.forName("org.duckdb.DuckDBDriver");
        try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
            for (String[] x : q) {
                String out;
                try (Statement s = c.createStatement(); ResultSet r = s.executeQuery(x[1])) {
                    r.next();
                    out = "VALUE " + r.getString(1);
                } catch (SQLException e) {
                    out = "ERROR " + e.getMessage().split("\n")[0];
                }
                System.out.println(x[0] + "\t" + x[1] + "\t" + out);
            }
        }
    }
}
