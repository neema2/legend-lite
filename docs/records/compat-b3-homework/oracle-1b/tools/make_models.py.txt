"""B3 step 1b, phase 1: writes the oracle's inputs from legend-engine 4.145.0's OWN test sources (read, never recalled).

  python3 -I make_models.py <legend_engine_src root> <out dir>

  models/calendar-h2.pure      the calendarAggregations tests' model (store, mapping, classes) re-packaged under
                               oracle1b::calendar, a LocalH2 connection whose testDataSetupSqls are the tests'

  models/sqlfn-h2.pure         the same for mapping::sqlFunction's log tests (oracle1b::sqlfn)
  models/*-duckdb-connection.pure  the DuckDB connection that replaces the H2 one (planoracle.SeedDuckDbMain)

  requests/sqlfn_log.tsv       the 2 log lambdas

"""

def pure_string(s):
    """A Pure single-quoted string literal."""
    return "'" + s.replace("\\", "\\\\").replace("'", "\\'") + "'"


def unescape(s):
    return s.replace("\\'", "'").replace("\\\\", "\\")


def setup_body(src, fn):
    start = src.index(fn)
    open_ = src.index("{", start)
    depth = 0
    for i in range(open_, len(src)):
        if src[i] == "{":
            depth += 1
        elif src[i] == "}":
            depth -= 1
            if depth == 0:
                return src[open_ + 1:i]
    raise ValueError(fn)


def execute_in_db_statements(body, variables):
    """Each executeInDb(<string expr>, $connection) of a setUp body, its string expression evaluated (literals and
    the body's `let` String variables joined by +)."""
    out = []
    for m in re.finditer(r"executeInDb\((.*?),\s*\$connection\);", body):
        expr = m.group(1)
        parts = re.findall(r"'((?:\\.|[^'\\])*)'|\$(\w+)", expr)
        text = ""
        for lit, var in parts:
            text += variables[var] if var else unescape(lit)
        out.append(text.strip())
    return out

# and DuckDB both read; Float/DOUBLE is an 8-byte double on both (lite's DuckDb.ddlType spells FLOAT and DOUBLE as
# DOUBLE). Statements that only one database reads are not used: the same list seeds legend-engine's H2 (LocalH2) and
# the DuckDB file (SeedDuckDbMain).
cal_seed = [
    "DROP TABLE IF EXISTS EmployeeTable;",
    "DROP TABLE IF EXISTS FirmTable;",
    "DROP TABLE IF EXISTS EmployeeDetailsTable;",
    "DROP SCHEMA IF EXISTS LegendCalendarSchema CASCADE;",
    "CREATE TABLE EmployeeTable (id INTEGER NOT NULL, hireDate DATE, hireType VARCHAR(10), fteFactor DOUBLE, PRIMARY KEY (id));",
]
emp = [s for s in cal_inserts if s.lower().startswith("insert into employeetable ")]
firm = [s for s in cal_inserts if s.lower().startswith("insert into firmtable ")]
det = [s for s in cal_inserts if s.lower().startswith("insert into employeedetailstable ")]
ny = [s for s in cal_inserts if s.lower().startswith("insert into legendcalendarschema.ny_calendar ")]
assert len(emp) + len(firm) + len(det) + len(ny) == len(cal_inserts), "an unclassified setUp statement"

cal_seed += firm
cal_seed += ["CREATE TABLE EmployeeDetailsTable (id INTEGER NOT NULL, birthDate DATE, yearsOfExperience DOUBLE, PRIMARY KEY (id));"]

cal_seed += ny
counts = (len(emp), len(firm), len(det), len(ny))
assert counts == (59, 2, 2, 60), counts

def seed_block(stmts):
    return ",\n".join("            " + pure_string(s) for s in stmts)
