import java.sql.*;
public class Sq { public static void main(String[] a) throws Exception {
 try (Connection c = DriverManager.getConnection("jdbc:sqlite::memory:"); Statement s = c.createStatement()) {
  java.io.BufferedReader r = new java.io.BufferedReader(new java.io.InputStreamReader(System.in));
  String l; while ((l = r.readLine()) != null) { if (l.isBlank()) continue;
   System.out.println(">> " + l);
   try (ResultSet rs = s.executeQuery(l)) { int n = rs.getMetaData().getColumnCount(); int k=0;
     while (rs.next() && k++<40) { StringBuilder b = new StringBuilder("   "); for (int i=1;i<=n;i++) b.append(rs.getString(i)).append(" | "); System.out.println(b); } }
   catch (Exception e) { System.out.println("   ERR " + e.getMessage()); } } } } }
