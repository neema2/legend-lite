SELECT ID, T.XS || ARRAY[9] AS LIST_CONCAT, T.XS || 7 AS LIST_APPEND, CASE WHEN T.N BETWEEN 1 AND CARDINALITY(T.XS) THEN T.XS[T.N] END AS LIST_GET FROM T

SELECT ID, (SELECT MIN(I) - 1 FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) AND T.XS[I] = 1) AS LIST_POSITION_0BASED FROM T

SELECT ID, ARRAY(SELECT ROW(T.XS[I], T.SS[I]) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= LEAST(CARDINALITY(T.XS), CARDINALITY(T.SS)) ORDER BY I) AS LIST_ZIP FROM T

SELECT ID, ARRAY(SELECT T.XS[I] FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) AND NOT EXISTS (SELECT 1 FROM SYSTEM_RANGE(1, 2147483647) R2(J) WHERE J < R.I AND T.XS[J] = T.XS[R.I]) ORDER BY I) AS LIST_DISTINCT_FIRST FROM T

SELECT ID, (SELECT SUM(T.XS[I]) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) S, (SELECT MIN(T.XS[I]) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) MN, (SELECT MAX(T.XS[I]) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) MX, (SELECT AVG(CAST(T.XS[I] AS DOUBLE)) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) AV, (SELECT MEDIAN(T.XS[I]) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) MED, (SELECT MODE(T.XS[I]) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) MO FROM T

SELECT ID, ARRAY(SELECT T.XS[I] FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) ORDER BY T.XS[I]) AS SORT_ASC, ARRAY(SELECT T.XS[I] FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) ORDER BY T.XS[I] DESC) AS SORT_DESC, ARRAY(SELECT T.XS[I] FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) ORDER BY I DESC) AS REV FROM T
;;
SELECT ID, ARRAY(SELECT T.XS[I] FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) AND I > 1 ORDER BY I) AS TAIL, ARRAY(SELECT T.XS[I] FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) AND I < CARDINALITY(T.XS) ORDER BY I) AS INIT, ARRAY(SELECT T.XS[I] FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0) AND I BETWEEN 2 AND 3 ORDER BY I) AS SLICE FROM T
;;
SELECT ID, ARRAY(SELECT X FROM SYSTEM_RANGE(0, 2147483647) R(X) WHERE X < T.N ORDER BY X) AS RANGE_N, ARRAY(SELECT 'v' FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= T.N) AS REPEAT_V FROM T
;;
SELECT ID, (SELECT BOOL_AND(T.XS[I] > 0) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) AS BAND, (SELECT BOOL_OR(T.XS[I] > 4) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) AS BOR, (SELECT EXP(SUM(LN(T.XS[I]))) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.XS), 0)) AS PRODUCT_FLOAT FROM T
;;
SELECT ID, (SELECT LISTAGG(T.SS[I], ',') WITHIN GROUP (ORDER BY I) FROM SYSTEM_RANGE(1, 2147483647) R(I) WHERE I <= COALESCE(CARDINALITY(T.SS), 0)) AS JOINED FROM T
