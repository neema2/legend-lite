import java.sql.*;
public class P {
    public static void main(String[] a) throws Exception {
        try (Connection c = DriverManager.getConnection("jdbc:duckdb:"); Statement s = c.createStatement()) {
            for (String q : new String[] {
                "select typeof(cast('3.14' as float)), cast('3.14' as float), cast(cast('3.14' as float) as double), cast('3.14' as float) = 3.14::double",
                "select typeof(cast('12' as BIGINT)), cast(' 12' as BIGINT)",
                "select cast('16777217' as float)::double, cast('16777217' as double)"}) {
                try (ResultSet r = s.executeQuery(q)) {
                    r.next(); StringBuilder b = new StringBuilder(q + "  =>  ");
                    for (int i = 1; i <= r.getMetaData().getColumnCount(); i++) b.append(r.getObject(i)).append(" | ");
                    System.out.println(b);
                }
            }
        }
    }
}
