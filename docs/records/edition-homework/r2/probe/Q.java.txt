import java.nio.file.*;
import java.sql.*;
import java.util.*;

/** Runs each line of a file as one SQL statement on a JDBC URL and prints "sql => value [type]" or "sql => ERROR msg". */
public class Q {
    public static void main(String[] a) throws Exception {
        String url = a[0];
        List<String> lines = Files.readAllLines(Path.of(a[1]));
        try (Connection c = DriverManager.getConnection(url)) {
            System.out.println("# " + c.getMetaData().getDatabaseProductName() + " " + c.getMetaData().getDatabaseProductVersion());
            for (String sql : lines) {
                if (sql.isBlank() || sql.startsWith("--")) continue;
                try (Statement s = c.createStatement()) {
                    boolean rs = s.execute(sql);
                    if (!rs) { System.out.println(sql + " => (ok)"); continue; }
                    try (ResultSet r = s.getResultSet()) {
                        StringBuilder sb = new StringBuilder();
                        int n = r.getMetaData().getColumnCount();
                        while (r.next()) {
                            for (int i = 1; i <= n; i++) {
                                Object o = r.getObject(i);
                                sb.append(o == null ? "NULL" : (o.getClass().isArray() ? Arrays.deepToString(new Object[]{o}) : o.toString()));
                                sb.append(" [").append(r.getMetaData().getColumnTypeName(i)).append("]");
                                if (i < n) sb.append(" | ");
                            }
                            sb.append(" ; ");
                        }
                        System.out.println(sql + " => " + sb);
                    }
                } catch (SQLException e) {
                    String m = e.getMessage().replace('\n', ' ');
                    System.out.println(sql + " => ERROR " + (m.length() > 160 ? m.substring(0, 160) : m));
                }
            }
        }
    }
}
