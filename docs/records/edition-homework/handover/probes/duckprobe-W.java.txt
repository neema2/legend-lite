import java.sql.*;
public class W {
  public static void main(String[] a) throws Exception {
    try (Connection c = DriverManager.getConnection("jdbc:duckdb:")) {
      String[] qs = {
        "SELECT TRY('{\"a\":\"1\"}'::JSON::VARIANT IS NOT DISTINCT FROM '{\"a\":1}'::JSON::VARIANT)",
        "SELECT TRY('{\"a\":1}'::JSON::VARIANT IS NOT DISTINCT FROM '[1]'::JSON::VARIANT)",
        "SELECT TRY('{\"b\":2,\"a\":1}'::JSON::VARIANT IS NOT DISTINCT FROM '{\"a\":1,\"b\":2}'::JSON::VARIANT)",
        "SELECT TRY('null'::JSON::VARIANT IS NOT DISTINCT FROM 'null'::JSON::VARIANT)",
        "SELECT TRY('{\"a\":null}'::JSON::VARIANT IS NOT DISTINCT FROM '{\"a\":1}'::JSON::VARIANT)",
        "SELECT TRY('{\"a\":1}'::JSON::VARIANT IS NOT DISTINCT FROM '{\"a\":1,\"b\":2}'::JSON::VARIANT)",
        "SELECT '{\"a\":\"1\"}'::JSON::VARIANT = '{\"a\":1}'::JSON::VARIANT",
        "SELECT '[1,\"x\"]'::JSON::VARIANT = '[1,2]'::JSON::VARIANT",
        "SELECT '{\"a\":1}'::JSON::VARIANT = '[1]'::JSON::VARIANT",
        "SELECT variant_typeof('{\"a\":1}'::JSON::VARIANT)",
        "SELECT variant_typeof('1'::JSON::VARIANT), variant_typeof('1.5'::JSON::VARIANT), variant_typeof('\"x\"'::JSON::VARIANT)",
        "SELECT '1'::JSON::VARIANT IS NOT DISTINCT FROM '1.0'::JSON::VARIANT",
        "SELECT 'null'::JSON::VARIANT = 'null'::JSON::VARIANT",
        "SELECT '{\"a\":1,\"b\":[1,2]}'::JSON::VARIANT::VARCHAR"
      };
      for (String q : qs) {
        try (Statement s = c.createStatement(); ResultSet r = s.executeQuery(q)) { r.next(); int n = r.getMetaData().getColumnCount(); StringBuilder b = new StringBuilder(); for (int i=1;i<=n;i++) b.append(r.getString(i)).append(" | "); System.out.println(q + " -> " + b); }
        catch (SQLException e) { System.out.println(q + " !! " + e.getMessage().split("\n")[0]); }
      }
    }
  }
}
