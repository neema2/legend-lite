import java.sql.*;
public class P {
  public static void main(String[] a) throws Exception {
    Connection c = DriverManager.getConnection("jdbc:h2:mem:p");
    String[] qs = {
      "SELECT JSON_ARRAYAGG(x ORDER BY y NULL ON NULL) AS l FROM (VALUES (1,2),(3,1)) t(x,y)",
      "SELECT * FROM (SELECT JSON_ARRAYAGG(x ORDER BY y NULL ON NULL) AS l FROM (VALUES (1,2),(3,1)) t(x,y)) AS d",
      "WITH c AS (SELECT x, y FROM (VALUES (1,2),(3,1)) t(x,y)) SELECT * FROM (SELECT JSON_ARRAYAGG(x ORDER BY y NULL ON NULL) AS l FROM c) AS d",
      "WITH c AS (SELECT x, y FROM (VALUES (1,2),(3,1)) t(x,y)) SELECT * FROM (SELECT JSON_ARRAYAGG(x ORDER BY y) AS l FROM c) AS d",
      "WITH c AS (SELECT x, y FROM (VALUES (1,2),(3,1)) t(x,y)) SELECT * FROM (SELECT ARRAY_AGG(x ORDER BY y) AS l FROM c) AS d",
      "WITH c AS (SELECT x, y FROM (VALUES (1,2),(3,1)) t(x,y)) SELECT * FROM (SELECT JSON_ARRAYAGG(x ORDER BY y ABSENT ON NULL) AS l FROM c) AS d"
    };
    for (String q : qs) {
      try (ResultSet r = c.createStatement().executeQuery(q)) { r.next(); System.out.println("OK  " + r.getObject(1) + "  <- " + q.substring(0, Math.min(90, q.length()))); }
      catch (Exception e) { System.out.println("ERR " + e.getMessage().split("\n")[0].substring(0, Math.min(140, e.getMessage().length())) + "  <- " + q.substring(0, Math.min(90,q.length()))); }
    }
  }
}
