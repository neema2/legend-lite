package com.legend.compiler.spec;

import com.legend.compiler.element.ModelContext;
import com.legend.compiler.element.type.Type;
import com.legend.compiler.spec.typed.TypedSpec;
import com.legend.protocol.spec.AppliedFunction;

import java.util.List;
import java.util.Optional;
import java.util.Set;

/** TEMPORARY MEASUREMENT PROBE (B3 step 5, not for commit): logs each comparison whose two sides' equality classes are
 *  known and differ. */
final class EqualityProbe {

    private EqualityProbe() {
    }

    private static final Set<String> PAIRS = Set.of("equal", "eq", "lessThan", "lessThanEqual", "greaterThan",
            "greaterThanEqual", "compare");
    private static final Set<String> NEEDLE_LAST = Set.of("in");
    private static final Set<String> COLLECTION_FIRST = Set.of("contains", "indexOf");

    static void probe(Application a, AppliedFunction af, ModelContext ctx) {
        String fqn = a.chosen().qualifiedName();
        String bare = fqn.substring(fqn.lastIndexOf(':') + 1);
        List<TypedSpec> args = a.args();
        if (args.size() != 2) {
            return;
        }
        boolean pair = PAIRS.contains(bare) || ((bare.equals("max") || bare.equals("min"))
                && args.get(0).info().multiplicity()
                        instanceof com.legend.compiler.element.type.Multiplicity.Bounded b && b.isToOne());
        if (!pair && !NEEDLE_LAST.contains(bare) && !COLLECTION_FIRST.contains(bare)) {
            return;
        }
        Type l = args.get(0).info().type();
        Type r = args.get(1).info().type();
        Optional<String> lc = of(l);
        Optional<String> rc = of(r);
        if (lc.isEmpty() || rc.isEmpty() || lc.get().equals(rc.get())) {
            return;
        }
        if (l instanceof Type.ClassType lt && r instanceof Type.ClassType rt
                && (ctx.isSubtype(lt.fqn(), rt.fqn()) || ctx.isSubtype(rt.fqn(), lt.fqn()))) {
            return;
        }
        String where = af.pos() == null ? "BUILT" : "WRITTEN";
        System.err.println("CROSSTYPE\t" + bare + "\t" + lc.get() + "\t" + rc.get() + "\t" + where + "\t" + af.pos());
    }

    static Optional<String> of(Type t) {
        return switch (t) {
            case Type.Primitive p -> switch (p) {
                case INTEGER, FLOAT, DECIMAL -> Optional.of(p.name());
                case STRING, BOOLEAN, BYTE, STRICT_DATE, DATE_TIME, STRICT_TIME -> Optional.of(p.name());
                case NUMBER, DATE, LATEST_DATE -> Optional.empty();
            };
            case Type.PrecisionDecimal d -> Optional.of("DECIMAL");
            case Type.EnumType e -> Optional.of("enum:" + e.fqn());
            case Type.ClassType c -> c.fqn().equals("meta::pure::metamodel::type::Any")
                    || c.fqn().endsWith("::Nil") || c.fqn().endsWith("::TDSNull") ? Optional.empty()
                    : Optional.of("class:" + c.fqn());
            default -> Optional.empty();
        };
    }
}
