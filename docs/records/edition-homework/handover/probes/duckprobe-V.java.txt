import java.sql.*;
public class V {
  public static void main(String[] a) throws Exception {
    try (Connection c = DriverManager.getConnection("jdbc:duckdb:"); Statement s = c.createStatement()) {
      String[] qs = {
        "SELECT version()",
        "SELECT '{\"a\":1,\"b\":2}'::JSON::VARIANT = '{\"b\":2,\"a\":1}'::JSON::VARIANT",
        "SELECT '{\"a\":1}'::JSON::VARIANT = '{\"a\":1.0}'::JSON::VARIANT",
        "SELECT '{\"a\":1}'::JSON::VARIANT = '{\"a\":2}'::JSON::VARIANT",
        "SELECT '[1,2]'::JSON::VARIANT = '[2,1]'::JSON::VARIANT",
        "SELECT '{\"a\":[1,{\"x\":\"y\",\"z\":null}]}'::JSON::VARIANT = '{\"a\":[1,{\"z\":null,\"x\":\"y\"}]}'::JSON::VARIANT",
        "SELECT '\"1\"'::JSON::VARIANT = '1'::JSON::VARIANT",
        "SELECT typeof('{\"a\":1}'::JSON::VARIANT)"
      };
      for (String q : qs) {
        try (ResultSet r = s.executeQuery(q)) { r.next(); System.out.println(q + " -> " + r.getString(1)); }
        catch (SQLException e) { System.out.println(q + " !! " + e.getMessage().split("\n")[0]); }
      }
    }
  }
}
