"""Render bucket 3's sorted classes as markdown (grouped by class, then by reason). argv[1] = r2 dir."""
import json, os, sys
from collections import OrderedDict, Counter

d = sys.argv[1]
rows = json.load(open(os.path.join(d, 'b3_sorted.json')))
ORDER = [('REF', '1. Refused by legend-engine (its own words)'),
         ('DBE', '2. A database error from legend-engine\'s SQL'),
         ('TB', '3. A legend-engine translation bug (a wrong value)'),
         ('PT', '4. A presentation or type difference'),
         ('LX', '5. Lite does something legend-engine deliberately does not (FLAGGED)'),
         ('OTH', '6. Other: legend-engine faults before or after its SQL (planner and run-time faults)')]
LIST_ORDER = ['LX', 'REF', 'DBE', 'TB', 'PT', 'OTH']
SUITES = ['Essential', 'Grammar', 'Relation', 'Standard', 'Unclassified']

def short(t):
    return t.replace('meta::pure::functions::', '')

out = []
cnt = Counter((r['suite'], r['class']) for r in rows)
out.append('| class | ' + ' | '.join(SUITES) + ' | total |')
out.append('|---|' + '---:|' * (len(SUITES) + 1))
for c, title in ORDER:
    vals = [cnt.get((s, c), 0) for s in SUITES]
    out.append('| ' + title.split(' (')[0] + ' | ' + ' | '.join(str(v) for v in vals) + ' | **' + str(sum(vals)) + '** |')
tot = [sum(cnt.get((s, c), 0) for c, _ in ORDER) for s in SUITES]
out.append('| **total** | ' + ' | '.join(str(v) for v in tot) + ' | **' + str(sum(tot)) + '** |')
out.append('')

for c in LIST_ORDER:
    title = dict(ORDER)[c]
    sel = [r for r in rows if r['class'] == c]
    out.append('### ' + title + ' -- ' + str(len(sel)))
    out.append('')
    groups = OrderedDict()
    for r in sel:
        groups.setdefault(r['why'], []).append(r)
    out.append('| # | tests | why | legend-engine manifest (relational-duckdb) |')
    out.append('|---:|---|---|---|')
    for why, rs in groups.items():
        names = '<br>'.join('`' + short(r['test']) + '` (' + r['suite'][:3] + ')' for r in rs)
        mans = ', '.join(r['manifest'] for r in rs)
        nums = ', '.join(str(r['i']) for r in rs)
        out.append('| ' + nums + ' | ' + names + ' | ' + why.replace('|', '\\|') + ' | ' + mans + ' |')
    out.append('')
open(os.path.join(d, 'b3_section.md'), 'w').write('\n'.join(out) + '\n')
print('\n'.join(out[:12]))
