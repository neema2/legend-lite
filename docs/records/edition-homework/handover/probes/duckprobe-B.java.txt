import java.sql.*;
public class B {
  public static void main(String[] a) throws Exception {
    String[] qs = {
      "SELECT epoch(TIMESTAMP_NS '2020-01-01 10:00:00.123456789')",
      "SELECT typeof(epoch(CAST(TIMESTAMP '2020-01-01' AS TIMESTAMP_NS)))",
    };
    for (String q : qs) {
      try (Connection c = DriverManager.getConnection("jdbc:duckdb:"); ResultSet r = c.createStatement().executeQuery(q)) { r.next(); System.out.println("OK  " + r.getObject(1) + " <- " + q); }
      catch (Exception e) { System.out.println("ERR " + e.getMessage().split("\n")[0] + " <- " + q); }
    }
  }
}
