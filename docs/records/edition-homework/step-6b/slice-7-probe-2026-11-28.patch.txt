+final class S7Sql {
+
+    private S7Sql() {
+    }
+
+    private static final ConcurrentHashMap<String, long[]> SIZES = new ConcurrentHashMap<>();
+    private static final ConcurrentHashMap<String, MessageDigest> DIGESTS = new ConcurrentHashMap<>();
+    private static final boolean FULL = System.getenv("S7PROBE_SQL_FULL") != null;
+    private static final StringBuilder TEXTS = new StringBuilder();
+
+    static void sent(String text) {
+        String test = System.getProperty("legend.test.current", "?");
+        long[] n = SIZES.computeIfAbsent(test, k -> new long[2]);
+        synchronized (n) {
+            n[0]++;
+            n[1] += text.length();
+            MessageDigest md = DIGESTS.computeIfAbsent(test, k -> {
+                try {
+                    return MessageDigest.getInstance("SHA-256");
+                } catch (java.security.NoSuchAlgorithmException e) {
+                    throw new IllegalStateException(e);
+                }
+            });
+            md.update(text.getBytes(StandardCharsets.UTF_8));
+            md.update((byte) 0);
+        }
+        if (FULL) {
+            synchronized (TEXTS) {
+                TEXTS.append("### ").append(test).append('\n').append(text).append('\n');
+            }
+        }
+    }
