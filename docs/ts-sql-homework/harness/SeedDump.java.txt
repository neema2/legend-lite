package com.legend.integration;

import java.sql.*;

/** Scratch: the docs test's seeded data, then a SQL file run on it, its rows as CSV to stdout. */
public final class SeedDump {
    public static void main(String[] a) throws Exception {
        var backend = com.legend.testbackend.TestBackends.platform();
        try (Connection c = backend.open()) {
            var m = TsDocumentsTest.class.getDeclaredMethod("seed", com.legend.testbackend.TestBackend.class, Connection.class);
            m.setAccessible(true);
            m.invoke(null, backend, c);
            String sql = java.nio.file.Files.readString(java.nio.file.Path.of(a[0]));
            for (int k = 0; k < 3; k++) {
                long t0 = System.nanoTime();
                int rows = 0;
                try (Statement s = c.createStatement(); ResultSet r = s.executeQuery(sql)) { while (r.next()) rows++; }
                System.err.println("TIME " + (System.nanoTime() - t0) / 1_000_000 + " ms, " + rows + " rows");
            }
            try (Statement s = c.createStatement(); ResultSet r = s.executeQuery(sql)) {
                int n = r.getMetaData().getColumnCount();
                StringBuilder b = new StringBuilder();
                for (int i = 1; i <= n; i++) b.append(i > 1 ? "," : "").append(r.getMetaData().getColumnLabel(i));
                System.out.println(b);
                while (r.next()) {
                    b.setLength(0);
                    for (int i = 1; i <= n; i++) { String v = r.getString(i); b.append(i > 1 ? "," : "").append(v == null ? "" : v); }
                    System.out.println(b);
                }
            }
        }
    }
}
