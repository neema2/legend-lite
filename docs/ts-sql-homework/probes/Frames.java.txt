import java.sql.*;
import java.time.*;
import java.util.*;

/** For one database: each spelling of a 6-month window frame against (d - 6 months, d], every business day 2020-2024. */
public class Frames {
    public static void main(String[] a) throws Exception {
        String url = a[0], user = a.length > 1 ? a[1] : null, pass = a.length > 2 ? a[2] : null;
        String inc = System.getProperty("inc"), mixed = System.getProperty("mixed"), edge = System.getProperty("edge");
        List<LocalDate> days = new ArrayList<>();
        for (LocalDate d = LocalDate.of(2020, 1, 2); !d.isAfter(LocalDate.of(2024, 12, 31)); d = d.plusDays(1)) {
            if (d.getDayOfWeek() != DayOfWeek.SATURDAY && d.getDayOfWeek() != DayOfWeek.SUNDAY) days.add(d);
        }
        Map<LocalDate, Long> truth = new HashMap<>();
        for (LocalDate d : days) {
            LocalDate lo = d.minusMonths(6);
            truth.put(d, days.stream().filter(x -> x.isAfter(lo) && !x.isAfter(d)).count());
        }
        try (Connection c = user == null ? DriverManager.getConnection(url) : DriverManager.getConnection(url, user, pass);
             Statement s = c.createStatement()) {
            s.execute("DROP TABLE IF EXISTS fr_days");
            if (System.getProperty("create") != null) {
                s.execute(System.getProperty("create"));
            } else {
                s.execute("CREATE TABLE fr_days (d DATE)");
                try (PreparedStatement p = c.prepareStatement("INSERT INTO fr_days VALUES (?)")) {
                    for (LocalDate d : days) { p.setObject(1, d); p.addBatch(); }
                    p.executeBatch();
                }
            }
            String[][] tries = {
                {"closed frame [d-6mo, d]", "count(*) OVER (ORDER BY d RANGE BETWEEN " + inc + " PRECEDING AND CURRENT ROW)"},
                {"'6 months less a day' frame", mixed == null ? null : "count(*) OVER (ORDER BY d RANGE BETWEEN " + mixed + " PRECEDING AND CURRENT ROW)"},
                {"closed frame minus the start date's rows", "count(*) OVER (ORDER BY d RANGE BETWEEN " + inc + " PRECEDING AND CURRENT ROW) - count(*) OVER (ORDER BY d RANGE BETWEEN " + (edge == null ? inc : edge) + " PRECEDING AND " + (edge == null ? inc : edge) + " PRECEDING)"},
            };
            for (String[] t : tries) {
                if (t[1] == null) { System.out.printf("  %-42s not spellable%n", t[0]); continue; }
                try (ResultSet r = s.executeQuery("SELECT d, " + t[1] + " AS n FROM fr_days")) {
                    int wrong = 0, n = 0;
                    while (r.next()) {
                        n++;
                        LocalDate d = r.getObject(1, LocalDate.class);
                        if (r.getLong(2) != truth.get(d)) wrong++;
                    }
                    System.out.printf("  %-42s %s%n", t[0], wrong == 0 ? "exact on all " + n + " dates" : wrong + " of " + n + " dates wrong");
                } catch (SQLException e) {
                    System.out.printf("  %-42s refused: %s%n", t[0], e.getMessage().lines().findFirst().orElse("").replaceAll("\\s+", " ").substring(0, Math.min(150, e.getMessage().lines().findFirst().orElse("").length())));
                }
            }
        }
    }
}
