import sys, re, collections
txt = open(sys.argv[1]).read()
events = txt.split('jdk.ExecutionSample {')[1:]
selfc = collections.Counter(); incl = collections.Counter(); n = 0
for e in events:
    m = re.search(r'stackTrace = \[(.*?)\n\s*\]', e, re.S)
    if not m: continue
    frames = [l.strip() for l in m.group(1).split('\n') if l.strip() and not l.strip().startswith('...')]
    frames = [re.sub(r'\s+line:.*', '', f) for f in frames]
    if not any('com.legend' in f for f in frames): continue
    n += 1
    selfc[frames[0]] += 1
    for f in set(f for f in frames if 'com.legend' in f):
        incl[f] += 1
print('samples', n)
print('--- self top 25')
for f, c in selfc.most_common(25): print(f'{100*c/n:5.1f}% {f}')
print('--- inclusive (com.legend) top 60')
for f, c in incl.most_common(60): print(f'{100*c/n:5.1f}% {f}')
