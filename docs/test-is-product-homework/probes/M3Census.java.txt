    /** Everything the census holds. */
    private static final class Store {
        final java.util.concurrent.ConcurrentHashMap<String, java.util.concurrent.atomic.LongAdder> counts =
                new java.util.concurrent.ConcurrentHashMap<>();
        final java.util.concurrent.ConcurrentHashMap<String, java.util.List<String>> examples =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** key -> every distinct value (tests, test classes). */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.Set<String>> sets =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9: "mode|form|sid|frame" -> the read sites (identity hashes of the nodes the hook rewrote). */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.Set<Integer>> stmtReads =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9: "mode|sid|frame" -> the hook instances that read it. */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.Set<Integer>> stmtHooks =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9 per program: "mode|bid|frame" -> "sid:identity" read sites. */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.Set<String>> bodyReads =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9: "mode|sid|frame" -> the test (for an example per bucket). */
        final java.util.concurrent.ConcurrentHashMap<String, String> readTest = new java.util.concurrent.ConcurrentHashMap<>();
        final java.util.concurrent.atomic.AtomicLong ids = new java.util.concurrent.atomic.AtomicLong();
        /** MV9 (tree): "mode|sid|frame" -> each resolved body's occurrence count of the frame (one per call). */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.List<Integer>> treeReads =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9 (tree): "mode|bid|frame" -> the total occurrences over the program's resolved bodies. */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.concurrent.atomic.LongAdder> treeBody =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9 (tree): "mode|sid|frame" -> the callers of each resolution holding the frame. */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.List<String>> treeCallers =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9 (tree): "mode|sid|frame" -> the forms its occurrences took (pasted, cte, cte(class extent)). */
        final java.util.concurrent.ConcurrentHashMap<String, java.util.Set<String>> treeForm =
                new java.util.concurrent.ConcurrentHashMap<>();
        /** MV9 (hook): "mode|bid|site" -> the statement that first rewrote the node. */
        final java.util.concurrent.ConcurrentHashMap<String, Long> firstSid = new java.util.concurrent.ConcurrentHashMap<>();
    }

        String t = TEST.get();
        if (t == null) {
            t = com.legend.lowering.StampCensus.CONTEXT.get();
        }
        return t == null ? "?" : t;
    }

    public static void count(String key) {
        S.counts.computeIfAbsent(key, k -> new java.util.concurrent.atomic.LongAdder()).increment();
    }

    public static void example(String key, String text) {
        java.util.List<String> l = S.examples.computeIfAbsent(key,
                k -> java.util.Collections.synchronizedList(new java.util.ArrayList<>()));
        if (l.size() < 3) {
            l.add(text.length() > 900 ? text.substring(0, 900) + "..." : text);
        }
    }

    /** {@code key} counted and its test recorded (distinct tests) with an example. */
    public static void counted(String key, String example) {
        count(key);
        S.sets.computeIfAbsent(key + " [tests]", k -> java.util.concurrent.ConcurrentHashMap.newKeySet()).add(test());
        example(key, test() + " :: " + example);
    }

    public static void member(String key, String value) {
        S.sets.computeIfAbsent(key, k -> java.util.concurrent.ConcurrentHashMap.newKeySet()).add(value);
    }

    public static String cls(Throwable e) {
        String m = e.getClass().getSimpleName() + ": " + String.valueOf(e.getMessage());
        m = m.replaceAll("[0-9]+", "N").replace('\n', ' ');
        return m.length() > 110 ? m.substring(0, 110) : m;
    }

    // ===== items 3 and 4: frame reads through the splice hook =====

    /** A hook application starts: its own hit map. */
    public static java.util.Map<String, String> pushHits() {
        java.util.Map<String, String> m = new java.util.LinkedHashMap<>();
        HITS.get().addLast(m);
        return m;
    }

    public static void popHits() {
        HITS.get().pollLast();
    }

    /** The frame {@code name} answered a read of the innermost hook application, in {@code form}. */
    public static void frameHit(String name, String form) {
        java.util.Map<String, String> m = HITS.get().peekLast();
        if (m != null) {
            m.putIfAbsent(name, form);
        }
    }

    /** A hook application that rewrote {@code n} after the frames in {@code hits} answered: one read of each. */
    public static void frameReads(long sid, long bid, String mode, int hook, java.util.Map<String, String> hits,
            Object n) {
        int site = System.identityHashCode(n);
        Long first = S.firstSid.putIfAbsent(mode + "|" + bid + "|" + site, sid);
        if (first != null && first != sid) {
            count("MV9.hook rewrites of a node an EARLIER statement rewrote (a let-prefix re-splice) " + mode);
        }
        for (var e : hits.entrySet()) {
            String k = mode + "|" + e.getValue() + "|" + sid + "|" + e.getKey();
            S.stmtReads.computeIfAbsent(k, x -> java.util.concurrent.ConcurrentHashMap.newKeySet()).add(site);
            S.stmtHooks.computeIfAbsent(mode + "|" + sid + "|" + e.getKey(),
                    x -> java.util.concurrent.ConcurrentHashMap.newKeySet()).add(hook);
            S.bodyReads.computeIfAbsent(mode + "|" + bid + "|" + e.getKey(),
                    x -> java.util.concurrent.ConcurrentHashMap.newKeySet()).add(sid + ":" + site);
            S.readTest.putIfAbsent(k, test());
            count("MV9.raw frame reads (hook rewrites) " + mode);
        }
    }

    /** MV9 (tree): a frame's chain was spliced; its root (the chain's source under its from) is remembered. */
    public static void spliced(String name, TypedSpec chain) {
        Object root = chain instanceof com.legend.compiler.spec.typed.TypedFrom fr ? fr.source() : chain;
        SPLICED.get().put(root, name);
    }

    /** MV9 (tree): {@code body}, about to be resolved and executed, scanned for the frames it reads: a pasted chain
     *  (its root's identity), a CTE reference ({@code TypedFrameRef}), a class extent over a CTE. */
    public static void treeReads(java.util.List<TypedSpec> body) {
        java.util.IdentityHashMap<Object, String> spliced = SPLICED.get();
        java.util.Map<String, Integer> found = new java.util.TreeMap<>();
        for (TypedSpec n : body) {
            scan(n, spliced, found);
        }
        if (found.isEmpty()) {
            return;
        }
        String mode = MODE.get();
        long sid = statement();
        long bid = body();
        String caller = StackWalker.getInstance().walk(st -> st.map(f -> f.getClassName().substring(
                f.getClassName().lastIndexOf('.') + 1) + "." + f.getMethodName())
                .filter(f -> !f.startsWith("M3Census.") && !f.startsWith("BodyPlanner.")).limit(2)
                .reduce((a, b) -> a + "<" + b).orElse("?"));
        found.forEach((nameForm, c) -> {
            String name = nameForm.substring(0, nameForm.indexOf('\u0001'));
            String form = nameForm.substring(nameForm.indexOf('\u0001') + 1);
            count("MV9T.per resolved body (one SQL) " + mode + " form=" + form + " occurrences="
                    + (c == 1 ? "1" : c == 2 ? "2" : "3+"));
            S.treeForm.computeIfAbsent(mode + "|" + sid + "|" + name,
                    x -> java.util.concurrent.ConcurrentHashMap.newKeySet()).add(form);
            S.treeCallers.computeIfAbsent(mode + "|" + sid + "|" + name,
                    x -> java.util.Collections.synchronizedList(new java.util.ArrayList<>())).add(caller);
            S.treeReads.computeIfAbsent(mode + "|" + sid + "|" + name,
                    x -> java.util.Collections.synchronizedList(new java.util.ArrayList<>())).add(c);
            S.treeBody.computeIfAbsent(mode + "|" + bid + "|" + name,
                    x -> new java.util.concurrent.atomic.LongAdder()).add(c);
            S.readTest.putIfAbsent("tree|" + mode + "|" + sid + "|" + name, test());
            String b = c == 1 ? "1" : c == 2 ? "2" : "3+";
            count("MV9T.per resolved body (one SQL) " + mode + " occurrences=" + b);
            if (c >= 2) {
                example("MV9T.per resolved body (one SQL) " + mode + " occurrences=" + b,
                        test() + " :: frame " + name + " x" + c);
            }
        });
    }

    private static void scan(TypedSpec n, java.util.IdentityHashMap<Object, String> spliced,
            java.util.Map<String, Integer> found) {
        String name = spliced.get(n);
        String form = "pasted";
        if (name == null && n instanceof com.legend.compiler.spec.typed.TypedFrameRef ref) {
            name = ref.name().startsWith("frame_") ? ref.name().substring("frame_".length()) : ref.name();
            form = "cte";
        }
        if (name == null && n instanceof com.legend.compiler.spec.typed.TypedFrom fr && fr.extentFrame() != null) {
            String ef = fr.extentFrame();
            name = ef.startsWith("frame_") ? ef.substring("frame_".length()) : ef;
            form = "cte(class extent)";
        }
        if (name != null) {
            found.merge(name + "\u0001" + form, 1, Integer::sum);
            return;
        }
        for (TypedSpec c : n.children()) {
            scan(c, spliced, found);
        }
    }

    // ===== item 6: an activity comment read =====

    /** Whether {@code n} (through the bodies of the user functions it calls) reads {@code .comment} of an activity. */
    public static boolean readsActivityComment(TypedSpec n, com.legend.compiler.spec.SpecCompiler specs,
            java.util.Map<com.legend.model.FunctionId, Boolean> memo) {
        if (n instanceof com.legend.compiler.spec.typed.TypedPropertyAccess pa && pa.property().equals("comment")
                && pa.source().info().type() instanceof com.legend.compiler.element.type.Type.ClassType ct
                && ct.fqn().endsWith("Activity")) {
            return true;
        }
        if (n instanceof com.legend.compiler.spec.typed.TypedUserCall uc) {
            com.legend.model.FunctionId key = uc.callee().id();
            Boolean known = memo.get(key);
            if (known == null) {
                memo.put(key, false);
                boolean r = false;
                try {
                    for (TypedSpec s : specs.compile(uc.callee()).body()) {
                        if (readsActivityComment(s, specs, memo)) {
                            r = true;
                            break;
                        }
                    }
                } catch (RuntimeException e) {
                    // an un-typeable callee: not decided here
                }
                memo.put(key, r);
                known = r;
            }
            if (known) {
                return true;
            }
        }
        for (TypedSpec c : n.children()) {
            if (readsActivityComment(c, specs, memo)) {
                return true;
            }
        }
        return false;
    }

    // ===== item 1: the outermost from()'s table renames =====

    public static void fromRenames(java.util.Map<String, String> fromTr, java.util.Map<String, String> envTr) {
        String cls;
        if (envTr.isEmpty()) {
            cls = "env carries none (the from()'s renames are not applied today)";
        } else {
            boolean all = true;
            boolean conflict = false;
            for (var e : fromTr.entrySet()) {
                String had = envTr.get(e.getKey());
                if (had == null) {
                    all = false;
                } else if (!had.equals(e.getValue())) {
                    conflict = true;
                }
            }
            cls = conflict ? "env carries a CONFLICTING rename"
                    : all ? "env already carries them (an execute frame's renames)"
                    : "env carries some other renames (part of the from()'s missing)";
        }
        String sid = Long.toString(statement());
        counted("M3h.prepares whose outermost from() carries renames", "from " + fromTr + " env " + envTr);
        counted("M3h.prepares: " + cls, "from " + fromTr + " env " + envTr);
        member("M3h.statements whose outermost from() carries renames [statements]", test() + "#" + sid);
        member("M3h.statements: " + cls + " [statements]", test() + "#" + sid);
    }

    /** The test class on this thread's stack (core's JUnit tests), or "?". */
    private static String testClass() {
        return StackWalker.getInstance().walk(st -> st.map(StackWalker.StackFrame::getClassName)
                .map(c -> c.indexOf('$') > 0 ? c.substring(0, c.indexOf('$')) : c)
                .filter(c -> c.endsWith("Test") || c.endsWith("Tests") || c.endsWith("IT"))
                .findFirst().orElse("?"));
    }

    private static boolean reaches(TypedSpec n, java.util.function.Predicate<TypedSpec> p,
            com.legend.compiler.spec.SpecCompiler specs, java.util.Map<com.legend.model.FunctionId, Boolean> memo) {
        if (p.test(n)) {
            return true;
        }
        if (n instanceof com.legend.compiler.spec.typed.TypedUserCall uc) {
            com.legend.model.FunctionId key = uc.callee().id();
            Boolean known = memo.get(key);
            if (known == null) {
                memo.put(key, false);
                boolean r = false;
                try {
                    for (TypedSpec s : specs.compile(uc.callee()).body()) {
                        if (reaches(s, p, specs, memo)) {
                            r = true;
                            break;
                        }
                    }
                } catch (RuntimeException e) {
                    // un-typeable: not decided here
                }
                memo.put(key, r);
                known = r;
            }
            if (known) {
                return true;
            }
        }
        for (TypedSpec c : n.children()) {
            if (reaches(c, p, specs, memo)) {
                return true;
            }
        }
        return false;
    }

    private static boolean seededRuntime(String rtFqn, com.legend.compiler.element.ModelContext ctx) {
        var rt = ctx.findRuntime(rtFqn);
        if (rt.isEmpty()) {
            return false;
        }
        for (var binding : rt.get().connectionBindings().entrySet()) {
            for (String connFqn : binding.getValue()) {
                var cd = ctx.findConnection(connFqn);
                if (cd.isPresent() && cd.get().specification() != null) {
                    var data = cd.get().specification().testData();
                    if (data != null && ((data.sqls() != null && !data.sqls().isEmpty()) || data.csv() != null)) {
                        return true;
                    }
                }
            }
        }
        return false;
    }

    /** The verdict functions {@code n} calls, through the user functions it calls (the path named). */
    private static void verdictCallees(TypedSpec n, com.legend.compiler.spec.SpecCompiler specs,
            java.util.Set<com.legend.model.FunctionId> seen, java.util.Set<String> out, String via) {
        String callee = n instanceof com.legend.compiler.spec.typed.TypedNativeCall nc ? nc.callee().qualifiedName()
                : n instanceof com.legend.compiler.spec.typed.TypedUserCall uc ? uc.callee().qualifiedName() : null;
        if (callee != null && com.legend.compiler.element.type.PlatformTypes.isVerdictFunction(callee)) {
            out.add(via + callee);
        }
        if (n instanceof com.legend.compiler.spec.typed.TypedUserCall uc && seen.add(uc.callee().id())) {
            try {
                for (TypedSpec st : specs.compile(uc.callee()).body()) {
                    verdictCallees(st, specs, seen, out, via + uc.callee().qualifiedName() + " > ");
                }
            } catch (RuntimeException e) {
                // un-typeable
            }
        }
        for (TypedSpec c : n.children()) {
            verdictCallees(c, specs, seen, out, via);
        }
    }

    /** The kinds of {@code s} the query door would refuse (B12 item 1), in the order the brief names them. */
    private static java.util.List<String> doorKinds(TypedSpec s, boolean last, com.legend.TypedQuery query,
            java.util.Map<com.legend.model.FunctionId, Boolean> effMemo,
            java.util.Map<com.legend.model.FunctionId, Boolean> verMemo,
            java.util.Map<com.legend.model.FunctionId, Boolean> aeMemo) {
        com.legend.compiler.spec.SpecCompiler specs = query.specs();
        java.util.List<String> kinds = new java.util.ArrayList<>();
        if (com.legend.compiler.spec.StatementEffects.containsEffect(s, specs, effMemo)) {
            kinds.add("effect");
        }
        java.util.List<com.legend.compiler.spec.typed.TypedFrom> froms =
                com.legend.compiler.spec.typed.ExecutionContext.froms(s);
        boolean inline = false;
        boolean named = false;
        for (var fr : froms) {
            if (!fr.sqlSetups().isEmpty() || !fr.csvSetups().isEmpty()) {
                inline = true;
            }
            var rt = fr.context().runtime();
            if (rt.isPresent() && seededRuntime(rt.get().fullPath(), query.context())) {
                named = true;
                member("Q.seeded named runtimes (the from()'s runtime vs the plan's)", rt.get().fullPath() + " vs "
                        + DOOR_RUNTIME.get());
            }
        }
        if (inline) {
            kinds.add("from() inline seeds");
        }
        if (named) {
            kinds.add("from()'s named runtime with seeds");
        }
        if (s instanceof com.legend.compiler.spec.typed.TypedLet let) {
            TypedSpec rhs = LetPlanner.rhs(let);
            if (rhs instanceof com.legend.compiler.spec.typed.TypedNativeCall ec
                    && (com.legend.builtin.NativeFn.Handle.isExecute(ec.callee().id())
                        || com.legend.builtin.NativeFn.Handle.of(ec.callee().id()).orElse(null)
                                == com.legend.builtin.NativeFn.Handle.EXECUTE_LEGEND_QUERY)) {
                kinds.add(last ? "let-bound execute (the LAST statement)" : "let-bound execute");
            }
        }
        boolean ae = reaches(s, n -> n instanceof com.legend.compiler.spec.typed.TypedNativeCall nc
                && com.legend.builtin.NativeFn.ContextOwner.of(nc.callee().id()).isPresent(), specs, aeMemo);
        boolean verdict = com.legend.compiler.spec.StatementEffects.callsVerdict(s, specs, verMemo);
        if (verdict) {
            kinds.add("verdict (callsVerdict)");
            java.util.Set<String> callees = new java.util.TreeSet<>();
            verdictCallees(s, specs, new java.util.HashSet<>(), callees, "");
            member("Q.verdict callees", callees.toString());
        }
        if (ae) {
            kinds.add("assertError");
        }
        return kinds;
    }

    /** Item 8: the runtime the product door plans for (its caller's argument). */
    public static final ThreadLocal<String> DOOR_RUNTIME = new ThreadLocal<>();

    /** Item 8: the product query door's typed body scanned (counted by kind, the test classes named). */
    public static @com.legend.base.Nullable String productDoor(com.legend.TypedQuery query) {
        try {
            java.util.List<TypedSpec> body = query.body();
            java.util.Map<com.legend.model.FunctionId, Boolean> effMemo = new java.util.HashMap<>();
            java.util.Map<com.legend.model.FunctionId, Boolean> verMemo = new java.util.HashMap<>();
            java.util.Map<com.legend.model.FunctionId, Boolean> aeMemo = new java.util.HashMap<>();
            String tc = testClass();
            count("Q.door queries");
            member("Q.door queries [classes]", tc);
            String first = null;
            java.util.Set<String> anyKinds = new java.util.LinkedHashSet<>();
            for (int i = 0; i < body.size(); i++) {
                TypedSpec s = body.get(i);
                count("Q.door statements");
                java.util.List<String> kinds = doorKinds(s, i == body.size() - 1, query, effMemo, verMemo, aeMemo);
                for (String k : kinds) {
                    count("Q.statements holding: " + k);
                    member("Q.statements holding: " + k + " [classes]", tc);
                    String src = String.valueOf(s);
                    example("Q.statements holding: " + k, tc + " :: " + (src.length() > 300 ? src.substring(0, 300)
                            : src));
                    anyKinds.add(k);
                }
                if (first == null && !kinds.isEmpty()) {
                    first = kinds.get(0);
                }
            }
            for (String k : anyKinds) {
                count("Q.queries holding: " + k);
            }
            if (first != null) {
                count("Q.queries refused (by the first statement's first kind): " + first);
                member("Q.queries refused (by the first statement's first kind): " + first + " [classes]", tc);
                count("Q.queries refused");
                member("Q.queries refused [classes]", tc);
            }
            return first;
        } catch (RuntimeException e) {
            count("Q.scan error " + cls(e));
            return null;
        }
    }

    // ===== the dump =====

    private static void histogram() {
        // per statement: each (mode, form, statement, frame) its number of read sites
        S.stmtReads.forEach((k, sites) -> {
            String[] p = k.split("\\|", 4);
            int n = sites.size();
            String b = n == 1 ? "1" : n == 2 ? "2" : "3+";
            String key = "MV9.per statement " + p[0] + " form=" + p[1] + " reads=" + b;
            count(key);
            count("MV9.per statement " + p[0] + " reads=" + b);
            if (n >= 2) {
                example(key, S.readTest.getOrDefault(k, "?") + " :: frame " + p[3] + " read " + n + "x");
            }
        });
        S.stmtHooks.forEach((k, hooks) -> {
            String[] p = k.split("\\|", 3);
            count("MV9.per statement " + p[0] + " hooks that read the frame=" + (hooks.size() >= 3 ? "3+"
                    : Integer.toString(hooks.size())));
        });
        S.bodyReads.forEach((k, sites) -> {
            String[] p = k.split("\\|", 3);
            int n = sites.size();
            count("MV9.per program " + p[0] + " reads=" + (n == 1 ? "1" : n == 2 ? "2" : "3+"));
        });
        S.treeReads.forEach((k, calls) -> {
            String[] p = k.split("\\|", 3);
            int total;
            int n;
            synchronized (calls) {
                total = calls.stream().mapToInt(Integer::intValue).sum();
                n = calls.size();
            }
            String b = total == 1 ? "1" : total == 2 ? "2" : "3+";
            count("MV9T.per statement " + p[0] + " occurrences (summed over its resolved bodies)=" + b);
            java.util.Set<String> forms = S.treeForm.getOrDefault(k, java.util.Set.of());
            count("MV9T.per statement " + p[0] + " form=" + (forms.size() == 1 ? forms.iterator().next()
                    : "mixed " + new java.util.TreeSet<>(forms)) + " occurrences=" + b);
            count("MV9T.per statement " + p[0] + " resolved bodies holding the frame=" + (n >= 3 ? "3+"
                    : Integer.toString(n)));
            if (n >= 2) {
                java.util.List<String> cs = S.treeCallers.getOrDefault(k, java.util.List.of());
                String sig;
                synchronized (cs) {
                    sig = String.join(" + ", cs);
                }
                count("MV9T.multi-body statement " + p[0] + " callers: " + sig);
                example("MV9T.multi-body statement " + p[0] + " callers: " + sig,
                        S.readTest.getOrDefault("tree|" + k, "?") + " :: frame " + p[2] + " " + calls);
            }
            if (total >= 2) {
                example("MV9T.per statement " + p[0] + " occurrences (summed over its resolved bodies)=" + b,
                        S.readTest.getOrDefault("tree|" + k, "?") + " :: frame " + p[2] + " " + calls);
            }
        });
        S.treeBody.forEach((k, v) -> {
            String[] p = k.split("\\|", 3);
            long t = v.sum();
            count("MV9T.per program " + p[0] + " occurrences=" + (t == 1 ? "1" : t == 2 ? "2" : "3+"));
        });
        S.sets.forEach((k, v) -> {
            S.counts.computeIfAbsent(k + " distinct", x -> new java.util.concurrent.atomic.LongAdder()).add(v.size());
        });
    }

    private static void dump() {
        histogram();
        StringBuilder sb = new StringBuilder();
        new java.util.TreeMap<>(S.counts).forEach((k, v) -> sb.append("[m3] ").append(k).append(" = ")
                .append(v.sum()).append('\n'));
        new java.util.TreeMap<>(S.examples).forEach((k, l) -> {
            synchronized (l) {
                for (String e : l) {
                    sb.append("[m3-ex] ").append(k).append(" :: ").append(e.replace('\n', ' ')).append('\n');
                }
            }
        });
        new java.util.TreeMap<>(S.sets).forEach((k, v) -> {
            if (k.endsWith("[classes]") || v.size() <= 60) {
                sb.append("[m3-set] ").append(k).append(" :: ").append(new java.util.TreeSet<>(v)).append('\n');
            }
        });
        System.err.print(sb);
        System.err.flush();
        try {
            java.nio.file.Path dir = java.nio.file.Path.of(DUMPS);
            java.nio.file.Files.createDirectories(dir);
            String label = System.getProperty("legend.judge.mode", "test") + "-"
                    + ProcessHandle.current().pid() + "-" + System.currentTimeMillis();
            java.nio.file.Files.writeString(dir.resolve(label + ".txt"), sb.toString());
        } catch (java.io.IOException | RuntimeException e) {
            System.err.println("[m3] dump file not written: " + e);
        }
    }
}
