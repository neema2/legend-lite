import java.sql.*;
public class H2Msg {
    public static void main(String[] a) throws Exception {
        try (Connection c = DriverManager.getConnection("jdbc:h2:mem:x"); Statement s = c.createStatement()) {
            s.execute("create table a10(i int);\ninsert into no_such_table values (1);");
        } catch (SQLException e) {
            System.out.println("[" + e.getMessage().replace("\n", "\\n") + "]");
            System.out.println("H2 dialect says: " + new com.legend.sql.dialect.H2().failingStatement(e.getMessage(),
                    java.util.List.of("create table a10(i int)", "insert into no_such_table values (1)")));
        }
    }
}
