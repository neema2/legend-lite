import re, statistics, sys
log = re.split(r'(?m)^### (?=(?:ON|OFF) r\d$)', open(sys.argv[1]).read())
skip = int(sys.argv[2])
rows = {'ON': [], 'OFF': []}
for ch in log[1:]:
    lab = ch.split('\n')[0]; v = lab.split()[0]
    ps = re.findall(r'pass (\d+): (\d+) of \d+ ok \(frameKept=(\w+)\); per query mean: plan ([\d.]+) us, run ([\d.]+) us, whole ([\d.]+) us; whole p50 ([\d.]+) us', ch)
    warm = [tuple(map(float, x[3:])) for x in ps if int(x[0]) > skip]
    rows[v].append(dict(lab=lab, kept=ps[0][2], plan=statistics.median(w[0] for w in warm), run=statistics.median(w[1] for w in warm),
                        whole=statistics.median(w[2] for w in warm), best=min(w[2] for w in warm), p50=statistics.median(w[3] for w in warm)))
for v in ('OFF', 'ON'):
    for r in rows[v]:
        print(f"{r['lab']:7s} frameKept={r['kept']:5s} per query (median over warm passes): plan {r['plan']:7.1f} us  run {r['run']:7.1f} us  whole {r['whole']:7.1f} us (best pass {r['best']:7.1f})  p50 {r['p50']:7.1f}")
for key, name in [('plan', 'plan (WASM planner, in-thread)'), ('run', 'run (runPlan on DuckDB-WASM)'), ('whole', 'whole query, mean'), ('best', 'whole query, best pass'), ('p50', 'whole query, p50')]:
    off = [r[key] for r in rows['OFF']]; on = [r[key] for r in rows['ON']]
    mo, mn = statistics.median(off), statistics.median(on)
    print(f"{name:32s} OFF {mo:7.1f} [{min(off):.1f}-{max(off):.1f}]  ON {mn:7.1f} [{min(on):.1f}-{max(on):.1f}]  ON-OFF {mn-mo:+6.1f} us ({100*(mn-mo)/mo:+.2f}%)")
