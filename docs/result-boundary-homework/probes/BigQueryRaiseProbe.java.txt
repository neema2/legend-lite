package org.finos.legend.lite.pct.extension;

import org.junit.jupiter.api.Test;

import java.sql.Connection;
import java.sql.ResultSet;
import java.sql.Statement;

/** A local measurement, never committed: which spelling of a projected raise BigQuery runs to its message. */
class BigQueryRaiseProbe {

    private static final String[][] CASES = {
        {"formatOfCastOfIf", "SELECT FORMAT_DATETIME('%Y', CAST(v AS DATETIME)) FROM (SELECT IF(TRUE, ERROR('raised'), CAST(NULL AS DATETIME)) AS v)"},
        {"formatOfIf", "SELECT FORMAT_DATETIME('%Y', v) FROM (SELECT IF(TRUE, ERROR('raised'), CAST(NULL AS DATETIME)) AS v)"},
        {"castOfIf", "SELECT CAST(v AS DATETIME) FROM (SELECT IF(TRUE, ERROR('raised'), CAST(NULL AS DATETIME)) AS v)"},
        {"castOfNull", "SELECT CAST(v AS DATETIME) FROM (SELECT CAST(NULL AS DATETIME) AS v)"},
        {"castOfDatetime", "SELECT CAST(v AS DATETIME) FROM (SELECT DATETIME '2020-01-02 03:04:05' AS v)"},
        {"formatOfCastOfIfLiteral", "SELECT FORMAT_DATETIME('%Y', CAST(v AS DATETIME)) FROM (SELECT IF(TRUE, ERROR('raised'), DATETIME '1970-01-01 00:00:00') AS v)"},
        {"inlineCastOfIf", "SELECT CAST(IF(TRUE, ERROR('raised'), CAST(NULL AS DATETIME)) AS DATETIME)"},
        {"formatOfCastOfCase", "SELECT FORMAT_DATETIME('%Y', CAST(v AS DATETIME)) FROM (SELECT CASE WHEN TRUE THEN ERROR('raised') ELSE CAST(NULL AS DATETIME) END AS v)"},
        {"formatOfCastOfIfOverRow", "SELECT FORMAT_DATETIME('%Y', CAST(v AS DATETIME)) FROM (SELECT IF(TRUE, ERROR('raised'), CAST(NULL AS DATETIME)) AS v FROM UNNEST([1]))"},
        {"formatOfCastOfBareError", "SELECT FORMAT_DATETIME('%Y', CAST(v AS DATETIME)) FROM (SELECT ERROR('raised') AS v)"},
        {"formatOfSafeCastOfIf", "SELECT FORMAT_DATETIME('%Y', SAFE_CAST(v AS DATETIME)) FROM (SELECT IF(TRUE, ERROR('raised'), CAST(NULL AS DATETIME)) AS v)"},
        {"planStatement", "SELECT CONCAT(IFNULL('value:Date[1]', ''), IFNULL(CAST(COALESCE(CONCAT(IFNULL('\\n', ''), IFNULL(CAST(STRING_AGG(`_pct1_a`.`_pct_line`, '\\n') AS STRING), '')), '') AS STRING), '')) AS `pctTds` FROM (   SELECT CASE WHEN `_pct1`.`value` IS NULL THEN 'null' ELSE CONCAT(IFNULL(CAST(FORMAT_DATETIME('%Y-%m-%dT%H:%M:%S', CAST(`_pct1`.`value` AS DATETIME)) AS STRING), ''), IFNULL(CAST(CASE WHEN RTRIM(CONCAT(SUBSTR(FORMAT_DATETIME('%E6S', CAST(`_pct1`.`value` AS DATETIME)), 4), '000'), '0') = '' THEN '' ELSE CONCAT(IFNULL('.', ''), IFNULL(CAST(RTRIM(CONCAT(SUBSTR(FORMAT_DATETIME('%E6S', CAST(`_pct1`.`value` AS DATETIME)), 4), '000'), '0') AS STRING), '')) END AS STRING), ''), IFNULL('+0000', '')) END AS `_pct_line`   FROM (     SELECT *     FROM (       SELECT IF(TRUE, ERROR(chr(31) || '1:2' || chr(30) || ('Invalid month: 13') || chr(31)), CAST(NULL AS DATETIME)) AS `value`     ) AS `_pct0`   ) AS `_pct1` ) AS `_pct1_a`"}
    };

    @Test
    void raises() throws Exception {
        try (Connection c = CloudLane.connect(CloudLane.Cloud.BIGQUERY); Statement st = c.createStatement()) {
            for (String[] k : CASES) {
                try (ResultSet rs = st.executeQuery(k[1])) {
                    rs.next();
                    System.out.println("[bq-raise] " + k[0] + " OK " + rs.getString(1));
                } catch (Exception e) {
                    String m = String.valueOf(e.getMessage()).replace("\n", " | ");
                    System.out.println("[bq-raise] " + k[0] + " ERROR " + m.substring(0, Math.min(m.length(), 300)));
                }
            }
        }
    }
}
