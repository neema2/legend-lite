package org.finos.legend.lite.pct.extension;

import org.junit.jupiter.api.Test;

import java.sql.Connection;
import java.sql.ResultSet;
import java.sql.ResultSetMetaData;
import java.sql.Statement;

/** A local measurement, never committed: Snowflake's text of a double, by each spelling. */
class SnowflakeTextProbe {
    @Test
    void floatTexts() throws Exception {
        String[] values = {"EXP(1)", "PI()", "1/3::FLOAT", "1.5e15::FLOAT", "2.5e-7::FLOAT", "1e300::FLOAT",
                "COSH(1.5::FLOAT)", "COSH(-100::FLOAT)", "123456789123456789::FLOAT", "0.1::FLOAT + 0.2::FLOAT",
                "3::FLOAT", "-1::FLOAT", "POWER(2.5::FLOAT, 5.7::FLOAT)", "ATAN(1.5::FLOAT)", "ACOS(2::FLOAT)"};
        String[] spellings = {"CAST(x AS VARCHAR)", "TO_VARCHAR(x)", "TO_VARCHAR(x, 'TME')", "TO_VARCHAR(x, 'TM9')",
                "TO_VARCHAR(x, 'TM')", "CAST(CAST(x AS NUMBER(38, 18)) AS VARCHAR)", "TO_JSON(TO_VARIANT(x))",
                "TO_VARCHAR(TO_VARIANT(x))", "SYSTEM$TYPEOF(x)"};
        try (Connection c = CloudLane.connect(CloudLane.Cloud.SNOWFLAKE); Statement st = c.createStatement()) {
            for (String v : values) {
                StringBuilder sql = new StringBuilder("SELECT x");
                for (String s : spellings) {
                    sql.append(", ").append(s);
                }
                sql.append(" FROM (SELECT ").append(v).append(" AS x)");
                StringBuilder line = new StringBuilder("[sf-probe] ").append(v).append(" |");
                try (ResultSet rs = st.executeQuery(sql.toString())) {
                    ResultSetMetaData md = rs.getMetaData();
                    rs.next();
                    line.append(" java=").append(rs.getDouble(1)).append(" |");
                    for (int i = 0; i < spellings.length; i++) {
                        line.append(" ").append(spellings[i]).append("=").append(rs.getString(i + 2)).append(" |");
                    }
                    line.append(" type=").append(md.getColumnTypeName(1));
                } catch (Exception e) {
                    line.append(" ERROR ").append(e.getMessage());
                }
                System.out.println(line);
            }
        }
    }
}
