# staged canon, correlated: three nested derived tables under a scalar subquery, the outer column read only innermost
SELECT v, (SELECT CASE WHEN regexp_full_match(t, '-?0+(\.0+)?') THEN '0.0' WHEN strpos(t, 'e') > 0 OR strpos(t, 'E') > 0 THEN concat(CASE WHEN starts_with(t, '-') THEN '-' ELSE '' END, CASE WHEN p <= 0 THEN concat('0.', rpad('', CAST(greatest(0 - p, 0) AS INTEGER), '0'), d) WHEN p >= length(d) THEN concat(d, rpad('', CAST(greatest(p - length(d), 0) AS INTEGER), '0'), '.0') ELSE concat(substr(d, 1, p), '.', substr(d, p + 1)) END) WHEN NOT strpos(t, '.') > 0 THEN concat(t, '.0') ELSE t END FROM (SELECT t, replace(m, '.', '') AS d, CASE WHEN strpos(m, '.') = 0 THEN length(m) ELSE strpos(m, '.') - 1 END + x AS p FROM (SELECT t, regexp_extract(t, '-?([0-9]+(?:\.[0-9]+)?)[eE]', 1) AS m, CAST(CASE WHEN regexp_extract(t, '[eE]([+-]?[0-9]+)$', 1) = '' THEN NULL ELSE regexp_extract(t, '[eE]([+-]?[0-9]+)$', 1) END AS INTEGER) AS x FROM (SELECT CAST(v AS VARCHAR) AS t) AS __f0) AS __f1) AS __f2) AS c FROM (SELECT CAST(1.3421e-08 AS DOUBLE) AS v UNION ALL SELECT CAST(1e300 AS DOUBLE) UNION ALL SELECT CAST(-2.5 AS DOUBLE) UNION ALL SELECT CAST(-0.0 AS DOUBLE) UNION ALL SELECT CAST(123456789012.0 AS DOUBLE)) AS s
